// Error handler utility
export function handleError(error: unknown, res: Response): void {
  console.error("Error:", error);

  // Handle Zod validation errors
  if (error instanceof ZodError) {
    const messages = error.errors.map(err => `${err.path.join('.')}: ${err.message}`);
    res.status(400).json({
      error: "Validation failed",
      details: messages,
    });
    return;
  }

  // Handle custom app errors
  if (error instanceof AppError) {
    res.status(error.statusCode).json({
      error: error.message,
    });
    return;
  }

  // Handle database errors
  if (error && typeof error === 'object' && 'code' in error) {
    const dbError = error as { code: string; detail?: string };
    
    // PostgreSQL error codes
    switch (dbError.code) {
      case '23505': // Unique violation
        res.status(409).json({
          error: "Duplicate entry",
          details: dbError.detail,
        });
        return;
      
      case '23503': // Foreign key violation
        res.status(400).json({
          error: "Referenced resource does not exist",
          details: dbError.detail,
        });
        return;
      
      case '23502': // Not null violation
        res.status(400).json({
          error: "Required field is missing",
          details: dbError.detail,
        });
        return;
    }
  }

  // Generic server error
  res.status(500).json({
    error: "Internal server error",
    ...(process.env.NODE_ENV === 'development' && {
      details: error instanceof Error ? error.message : String(error),
    }),
  });
}

// Async handler wrapper to catch errors
export function asyncHandler(
  fn: (req: any, res: Response, next: any) => Promise<any>
) {
  return (req: any, res: Response, next: any) => {
    Promise.resolve(fn(req, res, next)).catch((error) => {
      handleError(error, res);
    });
  };
}

import { Request, Response, NextFunction } from "express";

// Basic XSS protection - strips HTML tags and encodes special characters
function sanitizeString(value: string): string {
  if (typeof value !== 'string') return value;
  
  // Remove HTML tags
  let sanitized = value.replace(/<[^>]*>/g, '');
  
  // Encode special characters
  sanitized = sanitized
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#x27;')
    .replace(/\//g, '&#x2F;');
  
  // Trim whitespace
  return sanitized.trim();
}

// Recursively sanitize object properties
function sanitizeObject(obj: any): any {
  if (obj === null || obj === undefined) {
    return obj;
  }

  if (typeof obj === 'string') {
    return sanitizeString(obj);
  }

  if (Array.isArray(obj)) {
    return obj.map(item => sanitizeObject(item));
  }

  if (typeof obj === 'object') {
    const sanitized: any = {};
    for (const key in obj) {
      if (obj.hasOwnProperty(key)) {
        sanitized[key] = sanitizeObject(obj[key]);
      }
    }
    return sanitized;
  }

  return obj;
}

// Middleware to sanitize request body
export function sanitizeInput(req: Request, res: Response, next: NextFunction): void {
  if (req.body) {
    req.body = sanitizeObject(req.body);
  }

  if (req.query) {
    req.query = sanitizeObject(req.query);
  }

  if (req.params) {
    req.params = sanitizeObject(req.params);
  }

  next();
}

// Specific sanitization for text fields that should allow some formatting
export function sanitizeRichText(text: string): string {
  if (typeof text !== 'string') return text;
  
  // Allow line breaks but strip other HTML
  let sanitized = text.replace(/<(?!br\s*\/?)[^>]+>/g, '');
  
  // Encode special characters except line breaks
  sanitized = sanitized
    .replace(/&(?!#?\w+;)/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#x27;');
  
  return sanitized.trim();
}

// Rate limiting helper (for specific sensitive operations)
const requestCounts = new Map<string, { count: number; resetAt: number }>();

export function checkRateLimit(
  key: string,
  maxRequests: number,
  windowMs: number
): boolean {
  const now = Date.now();
  const record = requestCounts.get(key);

  if (!record || now > record.resetAt) {
    requestCounts.set(key, {
      count: 1,
      resetAt: now + windowMs,
    });
    return true;
  }

  if (record.count >= maxRequests) {
    return false;
  }

  record.count++;
  return true;
}

// Middleware for rate limiting specific routes
export function rateLimit(maxRequests: number = 100, windowMs: number = 60000) {
  return (req: Request, res: Response, next: NextFunction) => {
    const key = `${req.ip}-${req.path}`;
    
    if (!checkRateLimit(key, maxRequests, windowMs)) {
      return res.status(429).json({
        error: "Too many requests",
        message: "Please try again later",
      });
    }
    
    next();
  };
}

async createProgramWithStructure(
  programData: InsertProgram,
  structure: {
    weeks: Array<{
      weekNumber: number;
      notes?: string;
      days: Array<{
        dayNumber: number;
        name?: string;
        notes?: string;
        exercises: Array<InsertProgramExercise>;
      }>;
    }>;
  }
): Promise<{ program: Program; weeks: ProgramWeek[]; days: ProgramDay[]; exercises: ProgramExercise[] }> {
  return await db.transaction(async (tx) => {
    // Create program
    const [program] = await tx
      .insert(programs)
      .values(programData)
      .returning();

    const allWeeks: ProgramWeek[] = [];
    const allDays: ProgramDay[] = [];
    const allExercises: ProgramExercise[] = [];

    // Create weeks, days, and exercises
    for (const weekData of structure.weeks) {
      const [week] = await tx
        .insert(programWeeks)
        .values({
          programId: program.id,
          weekNumber: weekData.weekNumber,
          notes: weekData.notes,
        })
        .returning();
      allWeeks.push(week);

      for (const dayData of weekData.days) {
        const [day] = await tx
          .insert(programDays)
          .values({
            weekId: week.id,
            dayNumber: dayData.dayNumber,
            name: dayData.name,
            notes: dayData.notes,
          })
          .returning();
        allDays.push(day);

        for (const exerciseData of dayData.exercises) {
          const [exercise] = await tx
            .insert(programExercises)
            .values({
              ...exerciseData,
              dayId: day.id,
            })
            .returning();
          allExercises.push(exercise);
        }
      }
    }

    return { program, weeks: allWeeks, days: allDays, exercises: allExercises };
  });
}

  organizationId: string,

    // Get original program
    const [originalProgram] = await tx
      .select()
      .from(programs)
      .where(eq(programs.id, programId));

    if (!originalProgram) {
      throw new Error("Program not found");
    }

    // Create new program
    const [newProgram] = await tx
      .insert(programs)
      .values({
        name: newName,
        description: originalProgram.description,
        durationWeeks: originalProgram.durationWeeks,
        phase: originalProgram.phase,

        organizationId,
        isTemplate: originalProgram.isTemplate,
      })
      .returning();

    // Get all weeks
    const weeks = await tx
      .select()
      .from(programWeeks)
      .where(eq(programWeeks.programId, programId))
      .orderBy(programWeeks.weekNumber);

    // Copy weeks and days
    for (const week of weeks) {
      const [newWeek] = await tx
        .insert(programWeeks)
        .values({
          programId: newProgram.id,
          weekNumber: week.weekNumber,

        })
        .returning();

      // Get all days for this week
      const days = await tx
        .select()
        .from(programDays)

// Create complete workout session with all exercises and sets in one transaction
async createCompleteWorkoutSession(
  athleteId: string,
  sessionData: InsertWorkoutSession,
  exercises: Array<{
    exerciseId: string;
    order: number;
    notes?: string;
    sets: Array<InsertSetLog>;
  }>
): Promise<{
  session: WorkoutSession;
  exerciseLogs: Array<ExerciseLog & { sets: SetLog[] }>;
}> {
  return await db.transaction(async (tx) => {
    // Create session
    const [session] = await tx
      .insert(workoutSessions)
      .values({ ...sessionData, athleteId })
      .returning();

    const exerciseLogs: Array<ExerciseLog & { sets: SetLog[] }> = [];

    // Create exercise logs and sets
    for (const exerciseData of exercises) {
      const [exerciseLog] = await tx
        .insert(exerciseLogs)
        .values({
          sessionId: session.id,
          exerciseId: exerciseData.exerciseId,
          order: exerciseData.order,
          notes: exerciseData.notes,
        })
        .returning();

      const sets: SetLog[] = [];
      for (const setData of exerciseData.sets) {
        const [setLog] = await tx
          .insert(setLogs)
          .values({
            ...setData,
            exerciseLogId: exerciseLog.id,
          })
          .returning();
        sets.push(setLog);
      }

      exerciseLogs.push({ ...exerciseLog, sets });
    }

    return { session, exerciseLogs };
  });
}

  }

    .select({

    .from(exerciseLogs)
    .innerJoin(exercises, eq(exerciseLogs.exerciseId, exercises.id))
    .where(eq(exerciseLogs.sessionId, sessionId))
    .orderBy(exerciseLogs.order);

app.post('/api/programs/with-structure', isAuthenticated, async (req: AuthRequest, res) => {
  try {
    if (!req.currentUser) {
      return res.status(401).json({ message: "Unauthorized" });
    }

    const { program, structure } = req.body;

    if (!program || !structure) {
      return res.status(400).json({ message: "Program and structure are required" });
    }

    // Verify access to organization
    const org = await storage.getOrganization(program.organizationId);
    if (!org) {
      return res.status(404).json({ message: "Organization not found" });
    }

    const isOwner = org.ownerId === req.currentUser.id;
    const isCoach = req.currentUser.role === 'admin' || 
                   req.currentUser.role === 'head_coach' || 
                   req.currentUser.role === 'assistant_coach';

    if (!isOwner && !isCoach) {
      return res.status(403).json({ 
        message: "Forbidden: only coaches can create programs" 
      });
    }

    if (!isOwner) {
      const hasAccess = await hasOrganizationAccess(req.currentUser.id, program.organizationId);
      if (!hasAccess) {
        return res.status(403).json({ 
          message: "Forbidden: not a member of this organization" 
        });
      }
    }

    const result = await storage.createProgramWithStructure(
      { ...program, createdBy: req.currentUser.id },
      structure
    );

    res.json(result);
  } catch (error) {
    console.error("Error creating program with structure:", error);
    res.status(400).json({ message: "Failed to create program" });
  }
});
