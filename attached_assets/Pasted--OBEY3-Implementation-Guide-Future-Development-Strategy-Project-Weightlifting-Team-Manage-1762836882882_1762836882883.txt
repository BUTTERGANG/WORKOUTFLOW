## 1. IMPLEMENTATION STRATEGY

This guide synthesizes **OBEY.md** (critical production fixes) and **OBEY2.md** (architectural improvements) into a practical, step-by-step implementation plan.

1. **Fix Critical Bugs First** - Users can't use broken features
2. **Don't Break What Works** - Incremental changes with testing
3. **Deploy Often** - Small deployments reduce risk
4. **Measure Impact** - Track metrics before and after changes

```
Week 1-2:   Critical bug fixes (OBEY.md Priority 0)
```

**Problem:** Workout page crashes because endpoint doesn't exist.

**Priority:** P0 - BLOCKING

```typescript
// GET active program assignment for current user
app.get('/api/program-assignments/active', isAuthenticated, async (req: AuthRequest, res) => {
  try {
    const userId = req.currentUser!.id;

    logger.info('Fetching active program assignment', { userId });

    // Get the most recent assignment for this user
    const assignment = await db.query.programAssignments.findFirst({
      where: (assignments, { eq }) => eq(assignments.athleteId, userId),
      with: {
        program: {
          with: {
            weeks: {
              orderBy: (weeks, { asc }) => [asc(weeks.weekNumber)],
              with: {
                days: {
                  orderBy: (days, { asc }) => [asc(days.dayNumber)],
                  with: {
                    exercises: {
                      orderBy: (exercises, { asc }) => [asc(exercises.order)],
                      with: {
                        exercise: true,
                      },
                    },
                  },
                },
              },
            },
          },
        },
      },
      orderBy: (assignments, { desc }) => [desc(assignments.startDate)],
    });

    if (!assignment) {
      return res.status(404).json({
        message: "No active program assignment found"
      });
    }

    logger.info('Active program assignment found', {
      userId,
      assignmentId: assignment.id,
      programId: assignment.programId,
    });

    res.json(assignment);
  } catch (error: any) {
    logger.error("getting active program assignment", error);
    res.status(500).json({
      message: "Failed to get active program assignment"
    });
  }
});
```

### Fix 2: Missing `/api/workout-sessions/today` Endpoint

**Problem:** Can't load today's workout.

**Priority:** P0 - BLOCKING

**Step 1: Add to `server/routes.ts`**

```typescript
// GET today's workout session for current user
app.get('/api/workout-sessions/today', isAuthenticated, async (req: AuthRequest, res) => {
  try {
    const userId = req.currentUser!.id;
    const today = new Date();
    today.setHours(0, 0, 0, 0); // Start of today

    const tomorrow = new Date(today);
    tomorrow.setDate(tomorrow.getDate() + 1); // Start of tomorrow

    logger.info('Fetching today\'s workout session', { userId, date: today });

    // Find workout session for today
    const session = await db.query.workoutSessions.findFirst({
      where: (sessions, { eq, and, gte, lt }) => and(
        eq(sessions.athleteId, userId),
        gte(sessions.date, today),
        lt(sessions.date, tomorrow)
      ),
      with: {
        programDay: {
          with: {
            exercises: {
              orderBy: (exercises, { asc }) => [asc(exercises.order)],
              with: {
                exercise: true,
              },
            },
          },
        },
        exerciseLogs: {
          with: {
            sets: {
              orderBy: (sets, { asc }) => [asc(sets.setNumber)],
            },
            programExercise: {
              with: {
                exercise: true,
              },
            },
          },
        },
      },
    });

    if (!session) {
      logger.info('No workout session found for today', { userId, date: today });

      // Check if there's a scheduled workout from active program
      const assignment = await db.query.programAssignments.findFirst({
        where: (assignments, { eq }) => eq(assignments.athleteId, userId),
        with: {
          program: {
            with: {
              weeks: {
                orderBy: (weeks, { asc }) => [asc(weeks.weekNumber)],
                with: {
                  days: {
                    orderBy: (days, { asc }) => [asc(days.dayNumber)],
                    with: {
                      exercises: {
                        orderBy: (exercises, { asc }) => [asc(exercises.order)],
                        with: {
                          exercise: true,
                        },
                      },
                    },
                  },
                },
              },
            },
          },
        },
        orderBy: (assignments, { desc }) => [desc(assignments.startDate)],
      });

      if (!assignment) {
        return res.status(404).json({
          message: "No workout scheduled for today"
        });
      }

      // Calculate which day of the program we're on
      const startDate = new Date(assignment.startDate);
      startDate.setHours(0, 0, 0, 0);

      const daysSinceStart = Math.floor(
        (today.getTime() - startDate.getTime()) / (1000 * 60 * 60 * 24)
      );

      // Find the appropriate program day
      // Simple approach: cycle through weeks, find day by absolute day number
      let absoluteDayNumber = daysSinceStart % (assignment.program.durationWeeks * 7);
      let todaysProgramDay = null;
      let currentDayCount = 0;

      for (const week of assignment.program.weeks) {
        for (const day of week.days) {
          if (currentDayCount === absoluteDayNumber) {
            todaysProgramDay = day;
            break;
          }
          currentDayCount++;
        }
        if (todaysProgramDay) break;
      }

      if (!todaysProgramDay) {
        return res.status(404).json({
          message: "No workout scheduled for today"
        });
      }

      return res.json({
        scheduled: true,
        programDay: todaysProgramDay,
        assignment: assignment,
        suggestedStartTime: new Date(),
        message: "Workout scheduled but not started",
      });
    }

    logger.info('Workout session found', {
      userId,
      sessionId: session.id,
      status: session.status,
    });

    res.json(session);
  } catch (error: any) {
    logger.error("getting today's workout session", error);
    res.status(500).json({
      message: "Failed to get today's workout"
    });
  }
});
```

```typescript
app.post('/api/program-days/:dayId/exercises', isAuthenticated, async (req: AuthRequest, res) => {
  try {
    const dayId = req.params.dayId;
    const userId = req.currentUser!.id;

    logger.info('Adding exercise to program day', { dayId, userId });

    // Get the day to verify it exists and check access
    const day = await db.query.programDays.findFirst({
      where: (days, { eq }) => eq(days.id, parseInt(dayId)),
      with: {
        week: {
          with: {
            program: true,
          },
        },
      },
    });

    if (!day) {
      return res.status(404).json({ message: "Program day not found" });
    }

    const program = day.week.program;

    // Verify user has access to this program's organization
    const hasAccess = await hasOrganizationAccess(userId, program.organizationId);
    if (!hasAccess) {
      return res.status(403).json({
        message: "You don't have permission to modify this program"
      });
    }

    // Verify user is a coach
    const user = req.currentUser!;
    const isCoach = ['admin', 'head_coach', 'assistant_coach'].includes(user.role);
    if (!isCoach) {
      return res.status(403).json({
        message: "Only coaches can add exercises to programs"
      });
    }

    const data = insertProgramExerciseSchema.parse({
      ...req.body,
      dayId: parseInt(dayId),
    });

    const exercise = await storage.createProgramExercise(data);

    logger.info('Exercise added to program day', {
      dayId,
      exerciseId: exercise.id,
      userId,
    });

    res.json(exercise);
  } catch (error: any) {
    logger.error("adding exercise to program day", error);
    res.status(400).json({ message: "Failed to add exercise to program day" });
  }
});
```

### Step 2: Create Base Repository

**File: `server/repositories/base.repository.ts`**

```typescript
import { db } from '../db';
import { eq, and, or, desc, asc, SQL } from 'drizzle-orm';
import { PgTable } from 'drizzle-orm/pg-core';
import { logger } from '../logger';

export interface FindOptions {
  where?: SQL;
  orderBy?: SQL[];
  limit?: number;
  offset?: number;
}

export abstract class BaseRepository<T extends PgTable> {
  constructor(
    protected table: T,
    protected tableName: string
  ) {}

  async findById(id: string | number): Promise<any | null> {
    try {
      const result = await db.query[this.tableName].findFirst({
        where: (table: any, { eq }: any) => eq(table.id, id),
      });
      return result || null;
    } catch (error) {
      logger.error(`Error finding ${this.tableName} by id`, { id, error });
      throw error;
    }
  }

  async findMany(options: FindOptions = {}): Promise<any[]> {
    try {
      let query = db.select().from(this.table);

      if (options.where) {
        query = query.where(options.where) as any;
      }

      if (options.orderBy && options.orderBy.length > 0) {
        query = query.orderBy(...options.orderBy) as any;
      }

      if (options.limit) {
        query = query.limit(options.limit) as any;
      }

      if (options.offset) {
        query = query.offset(options.offset) as any;
      }

      return await query;
    } catch (error) {
      logger.error(`Error finding ${this.tableName}`, { options, error });
      throw error;
    }
  }

  async findFirst(where: SQL): Promise<any | null> {
    try {
      const results = await this.findMany({ where, limit: 1 });
      return results[0] || null;
    } catch (error) {
      logger.error(`Error finding first ${this.tableName}`, { where, error });
      throw error;
    }
  }

  async create(data: any): Promise<any> {
    try {
      const [result] = await db.insert(this.table).values(data).returning();
      logger.info(`${this.tableName} created`, { id: result.id });
      return result;
    } catch (error) {
      logger.error(`Error creating ${this.tableName}`, { data, error });
      throw error;
    }
  }

  async update(id: string | number, data: any): Promise<any> {
    try {
      const [result] = await db
        .update(this.table)
        .set({ ...data, updatedAt: new Date() })
        .where(eq((this.table as any).id, id))
        .returning();

      logger.info(`${this.tableName} updated`, { id });
      return result;
    } catch (error) {
      logger.error(`Error updating ${this.tableName}`, { id, data, error });
      throw error;
    }
  }

  async delete(id: string | number): Promise<void> {
    try {
      await db.delete(this.table).where(eq((this.table as any).id, id));
      logger.info(`${this.tableName} deleted`, { id });
    } catch (error) {
      logger.error(`Error deleting ${this.tableName}`, { id, error });
      throw error;
    }
  }

  async count(where?: SQL): Promise<number> {
    try {
      let query = db.select({ count: sql<number>`count(*)` }).from(this.table);

      if (where) {
        query = query.where(where) as any;
      }

      const [result] = await query;
      return result.count;
    } catch (error) {
      logger.error(`Error counting ${this.tableName}`, { where, error });
      throw error;
    }
  }

  async exists(where: SQL): Promise<boolean> {
    const count = await this.count(where);
    return count > 0;
  }
}
```

```typescript
import { UserRepository } from '../repositories/user.repository';
import { OrganizationRepository } from '../repositories/organization.repository';
import { TeamRepository } from '../repositories/team.repository';
import { ProgramRepository } from '../repositories/program.repository';
import { NotFoundError, ForbiddenError } from '../errors';
import { logger } from '../logger';
import type { User, Organization, Team, Program } from '../../shared/schema';

export class AuthorizationService {
  constructor(
    private userRepo: UserRepository,
    private orgRepo: OrganizationRepository,
    private teamRepo: TeamRepository,
    private programRepo: ProgramRepository
  ) {}

  // User operations
  async getUser(userId: string): Promise<User> {
    const user = await this.userRepo.findById(userId);
    if (!user) {
      throw new NotFoundError('User not found');
    }
    return user;
  }

  // Role checks
  verifyAdminRole(user: User): void {
    if (user.role !== 'admin') {
      throw new ForbiddenError('This action requires admin privileges');
    }
  }

  verifyCoachRole(user: User): void {
    const coachRoles = ['admin', 'head_coach', 'assistant_coach'];
    if (!coachRoles.includes(user.role)) {
      throw new ForbiddenError('This action requires coach privileges');
    }
  }

  verifyHeadCoachRole(user: User): void {
    const headCoachRoles = ['admin', 'head_coach'];
    if (!headCoachRoles.includes(user.role)) {
      throw new ForbiddenError('This action requires head coach privileges');
    }
  }

  isCoach(user: User): boolean {
    return ['admin', 'head_coach', 'assistant_coach'].includes(user.role);
  }

  isHeadCoach(user: User): boolean {
    return ['admin', 'head_coach'].includes(user.role);
  }

  isAdmin(user: User): boolean {
    return user.role === 'admin';
  }

  // Organization authorization
  async verifyOrganizationOwnership(
    userId: string,
    orgId: string
  ): Promise<Organization> {
    const org = await this.orgRepo.findById(orgId);

    if (!org) {
      throw new NotFoundError('Organization not found');
    }

    if (org.ownerId !== userId) {
      throw new ForbiddenError(
        'You must be the organization owner to perform this action'
      );
    }

    logger.debug('Organization ownership verified', { userId, orgId });
    return org;
  }

  async verifyOrganizationMembership(
    userId: string,
    orgId: string
  ): Promise<Organization> {
    const org = await this.orgRepo.findById(orgId);

    if (!org) {
      throw new NotFoundError('Organization not found');
    }

    // Owner is always a member
    if (org.ownerId === userId) {
      return org;
    }

    // Check membership
    const membership = await this.orgRepo.findMembership(orgId, userId);
    if (!membership) {
      throw new ForbiddenError(
        'You must be a member of this organization to perform this action'
      );
    }

    logger.debug('Organization membership verified', { userId, orgId });
    return org;
  }

  async verifyOrganizationCoachAccess(
    userId: string,
    orgId: string
  ): Promise<Organization> {
    const user = await this.getUser(userId);
    this.verifyCoachRole(user);

    return await this.verifyOrganizationMembership(userId, orgId);
  }

  // Team authorization
  async verifyTeamAccess(userId: string, teamId: string): Promise<Team> {
    const team = await this.teamRepo.findById(teamId);

    if (!team) {
      throw new NotFoundError('Team not found');
    }

    // Verify user has access to team's organization
    await this.verifyOrganizationMembership(userId, team.organizationId);

    // Verify user is a member of the team
    const membership = await this.teamRepo.findMembership(teamId, userId);
    if (!membership) {
      throw new ForbiddenError('You are not a member of this team');
    }

    logger.debug('Team access verified', { userId, teamId });
    return team;
  }

  async verifyTeamCoachAccess(userId: string, teamId: string): Promise<Team> {
    const team = await this.verifyTeamAccess(userId, teamId);
    const user = await this.getUser(userId);

    // Verify user has coach role
    this.verifyCoachRole(user);

    logger.debug('Team coach access verified', { userId, teamId });
    return team;
  }

  // Program authorization
  async verifyProgramAccess(userId: string, programId: string): Promise<Program> {
    const program = await this.programRepo.findById(programId);

    if (!program) {
      throw new NotFoundError('Program not found');
    }

    // Verify user has access to program's organization
    await this.verifyOrganizationMembership(userId, program.organizationId);

    logger.debug('Program access verified', { userId, programId });
    return program;
  }

  async verifyProgramModifyAccess(
    userId: string,
    programId: string
  ): Promise<Program> {
    const program = await this.verifyProgramAccess(userId, programId);
    const user = await this.getUser(userId);

    // Only coaches or program creator can modify
    if (program.createdBy !== userId && !this.isCoach(user)) {
      throw new ForbiddenError(
        'Only coaches or the program creator can modify programs'
      );
    }

    logger.debug('Program modify access verified', { userId, programId });
    return program;
  }

  // Helper: Check if user can manage another user
  canManageUser(actor: User, target: User): boolean {
    // Admins can manage anyone
    if (actor.role === 'admin') return true;

    // Head coaches can manage assistant coaches and athletes
    if (actor.role === 'head_coach') {
      return ['assistant_coach', 'athlete'].includes(target.role);
    }

    // Assistant coaches can't manage anyone
    return false;
  }

  // Get effective permissions
  async getUserPermissions(userId: string, orgId: string) {
    const org = await this.verifyOrganizationMembership(userId, orgId);
    const user = await this.getUser(userId);

    const isOwner = org.ownerId === userId;
    const isAdmin = user.role === 'admin';
    const isHeadCoach = user.role === 'head_coach';
    const isCoach = this.isCoach(user);

    return {
      canCreatePrograms: isCoach,
      canDeletePrograms: isOwner || isAdmin,
      canManageTeams: isCoach,
      canInviteMembers: isCoach,
      canManageOrganization: isOwner || isAdmin,
      canAssignPrograms: isCoach,
      canViewAllWorkouts: isCoach,
      canModifySettings: isOwner,
      canDeleteOrganization: isOwner,
    };
  }
}
```

```typescript
    logger.info('Program updated', { programId, userId });
    return updated;
  }

    const program = await this.programRepo.findById(programId);

    if (!program) {
      throw new NotFoundError('Program not found');
    }

    // Only creator or org owner can delete
    const isCreator = program.createdBy === userId;
    const org = await this.orgRepo.findById(program.organizationId);
    const isOrgOwner = org?.ownerId === userId;

    if (!isCreator && !isOrgOwner) {

    const existingWeeks = await this.programRepo.findWeeksByProgram(programId);
```

```typescript
  if (error.name === 'ForbiddenError') {
    return res.status(403).json({ message: error.message });
  }

  if (error.name === 'ValidationError') {
    return res.status(400).json({ message: error.message });

  logger.error('Unexpected service error', error);
  res.status(500).json({ message: 'Internal server error' });
```

```typescript
export function usePrograms(organizationId: string | undefined) {
  return useQuery<Program[]>({
    queryKey: ['/api/programs', { organizationId }],
    enabled: !!organizationId,
    staleTime: 5 * 60 * 1000, // 5 minutes
    select: (data) => {
      // Transform/sort data if needed
      return data.sort((a, b) =>
        new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
      );
    },
  });
}
```

```typescript
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { useToast } from '@/hooks/use-toast';
import { apiRequest } from '@/lib/queryClient';

interface CreateProgramInput {
  name: string;
  description?: string;
  durationWeeks: number;
  phase: string;
  organizationId: string;
}

export function useCreateProgram() {
  const queryClient = useQueryClient();
  const { toast } = useToast();

  return useMutation({
    mutationFn: (data: CreateProgramInput) =>
      apiRequest('/api/programs', { method: 'POST', body: data }),

    onSuccess: (newProgram, variables) => {
      // Optimistically update cache
      queryClient.setQueryData<any[]>(
        ['/api/programs', { organizationId: variables.organizationId }],
        (old) => (old ? [newProgram, ...old] : [newProgram])
      );

      toast({
        title: 'Program created!',
        description: `${newProgram.name} has been created successfully.`,
      });
    },

    onError: (error: Error) => {
      toast({
        title: 'Failed to create program',
        description: error.message,
        variant: 'destructive',
      });
    },
  });
}
```

```typescript
        </div>

        {user?.role !== 'athlete' && (
          <Button onClick={() => setCreateDialogOpen(true)}>
            <Plus className="mr-2 h-4 w-4" />
            Create Program
          </Button>
        )}
      </div>

      {/* Programs list */}
      <ProgramsList
        programs={programs}
        isLoading={isLoading}
        error={error as Error}
        onSelectProgram={setSelectedProgram}
        onDeleteProgram={handleDeleteProgram}
        onAssignProgram={handleAssignProgram}
        showActions={user?.role !== 'athlete'}
      />

      {/* Dialogs */}
      <CreateProgramDialog
        open={createDialogOpen}
        onOpenChange={setCreateDialogOpen}
        organizationId={currentOrganization.id}
      />

      <AssignProgramDialog
        open={assignDialogOpen}
        onOpenChange={setAssignDialogOpen}
        program={programToAssign}
        organizationId={currentOrganization.id}
      />
    </div>
  );
}
```

**Result:** Main page went from 1,323 lines to ~150 lines!

### Monitoring & Observability

**Add Sentry for Error Tracking**

```bash
npm install @sentry/node @sentry/react
```

**Server setup (`server/monitoring/sentry.ts`):**

```typescript
import * as Sentry from '@sentry/node';
import type { Express } from 'express';

export function initSentry(app: Express) {
  if (!process.env.SENTRY_DSN) {
    console.warn('SENTRY_DSN not set - error tracking disabled');
    return;
  }

  Sentry.init({
    dsn: process.env.SENTRY_DSN,
    environment: process.env.NODE_ENV || 'development',
    tracesSampleRate: process.env.NODE_ENV === 'production' ? 0.1 : 1.0,

    beforeSend(event, hint) {
      // Remove sensitive data
      if (event.request?.headers) {
        delete event.request.headers['authorization'];
        delete event.request.headers['cookie'];
      }

      if (event.request?.data && typeof event.request.data === 'object') {
        delete event.request.data.password;
        delete event.request.data.passwordHash;
      }

      return event;
    },
  });

  // Request handler must be first
  app.use(Sentry.Handlers.requestHandler());
  app.use(Sentry.Handlers.tracingHandler());
}

export function setupSentryErrorHandler(app: Express) {
  // Must be before other error handlers
  app.use(Sentry.Handlers.errorHandler());
}
```

**Add to `server/index.ts`:**

```typescript

// Initialize Sentry FIRST
initSentry(app);

// ... middleware ...

// ... routes ...

// Setup Sentry error handler
setupSentryErrorHandler(app);

// ... your error handlers ...
```

### Migration Guide: From Context API to Zustand

**Before:**

```typescript
// contexts/AppContext.tsx

// Stores full object in localStorage
localStorage.setItem('currentOrganization', JSON.stringify(org));
```

**After:**

```typescript
// stores/app.store.ts
export const useAppStore = create(
  persist(
    (set) => ({
      selectedOrgId: null,
      setSelectedOrganization: (id) => set({ selectedOrgId: id }),
    }),
    { name: 'app-state' }
  )
);

// hooks/useCurrentOrganization.ts
export function useCurrentOrganization() {
  const orgId = useAppStore(state => state.selectedOrgId);

  return useQuery({
    queryKey: ['/api/organizations', orgId],
    enabled: !!orgId,
  });
}
```

1. Create Zustand store
