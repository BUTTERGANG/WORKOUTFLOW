def get_category_id(cursor, category_name):
    """Get the ID for a category name"""
    cursor.execute("SELECT id FROM exercise_categories WHERE name = %s", (category_name,))
    result = cursor.fetchone()
    return result[0] if result else None

def get_tag_id(cursor, tag_name):
    """Get the ID for a secondary tag name"""
    if tag_name is None:
        return None
    cursor.execute("SELECT id FROM exercise_secondary_tags WHERE name = %s", (tag_name,))
    result = cursor.fetchone()
    return result[0] if result else None

def populate_exercises(conn):
    """Populate the exercises table with all 193 exercises"""
    cursor = conn.cursor()
    
    inserted_count = 0
    skipped_count = 0
    
    for exercise in EXERCISES:
        try:
            # Get category and tag IDs
            primary_id = get_category_id(cursor, exercise['primary'])
            secondary_id = get_tag_id(cursor, exercise.get('secondary'))
            
            # Check if exercise already exists
            cursor.execute("SELECT id FROM exercises WHERE name = %s", (exercise['name'],))
            if cursor.fetchone():
                print(f"Skipping existing exercise: {exercise['name']}")
                skipped_count += 1
                continue
            
            # Insert exercise
            cursor.execute("""
                INSERT INTO exercises (
                    name, 
                    primary_category_id, 
                    secondary_tag_id,
                    youtube_url,
                    is_competition_lift
                ) VALUES (%s, %s, %s, %s, %s)
                RETURNING id
            """, (
                exercise['name'],
                primary_id,
                secondary_id,
                exercise.get('youtube', ''),
                exercise.get('is_competition', False)
            ))
            
            exercise_id = cursor.fetchone()[0]
            inserted_count += 1
            print(f"Inserted: {exercise['name']} (ID: {exercise_id})")
            
        except Exception as e:
            print(f"Error inserting {exercise['name']}: {e}")
            conn.rollback()
            continue
    
    conn.commit()
    cursor.close()
    
    print(f"\n✅ Completed!")
    print(f"   Inserted: {inserted_count}")
    print(f"   Skipped: {skipped_count}")
    print(f"   Total: {len(EXERCISES)}")

def main():
    """Main function to connect and populate database"""
    try:
        # Connect to database
        conn = psycopg2.connect(**DB_CONFIG)
        print("Connected to database successfully!")
        
        # Populate exercises
        populate_exercises(conn)
        
        # Close connection
        conn.close()
        print("\nDatabase connection closed.")
        
    except psycopg2.Error as e:
        print(f"Database error: {e}")
    except Exception as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    main()
