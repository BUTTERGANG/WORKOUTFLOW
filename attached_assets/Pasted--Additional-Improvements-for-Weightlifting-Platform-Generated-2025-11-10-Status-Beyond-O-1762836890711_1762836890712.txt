```typescript
// components/SmartInput.tsx
import { useState, useEffect } from 'react';
import { Input } from '@/components/ui/input';
import { Check, X, Loader2 } from 'lucide-react';
import { cn } from '@/lib/utils';

interface SmartInputProps {
  value: string;
  onChange: (value: string) => void;
  validate?: (value: string) => Promise<{ valid: boolean; message?: string }>;
  debounceMs?: number;
}

export function SmartInput({
  value,
  onChange,
  validate,
  debounceMs = 500,
}: SmartInputProps) {
  const [validationState, setValidationState] = useState<
    'idle' | 'validating' | 'valid' | 'invalid'
  >('idle');
  const [validationMessage, setValidationMessage] = useState<string>('');

  useEffect(() => {
    if (!validate || !value) {
      setValidationState('idle');
      return;
    }

    const timer = setTimeout(async () => {
      setValidationState('validating');
      try {
        const result = await validate(value);
        setValidationState(result.valid ? 'valid' : 'invalid');
        setValidationMessage(result.message || '');
      } catch (error) {
        setValidationState('invalid');
        setValidationMessage('Validation failed');
      }
    }, debounceMs);

    return () => clearTimeout(timer);
  }, [value, validate, debounceMs]);

  return (
    <div className="relative">
      <Input
        value={value}
        onChange={(e) => onChange(e.target.value)}
        className={cn(
          validationState === 'valid' && 'border-green-500',
          validationState === 'invalid' && 'border-red-500'
        )}
      />
      <div className="absolute right-3 top-1/2 -translate-y-1/2">
        {validationState === 'validating' && (
          <Loader2 className="h-4 w-4 animate-spin text-muted-foreground" />
        )}
        {validationState === 'valid' && (
          <Check className="h-4 w-4 text-green-500" />
        )}
        {validationState === 'invalid' && (
          <X className="h-4 w-4 text-red-500" />
        )}
      </div>
      {validationMessage && (
        <p className={cn(
          'text-sm mt-1',
          validationState === 'valid' ? 'text-green-600' : 'text-red-600'
        )}>
          {validationMessage}
        </p>
      )}
    </div>
  );
}

// Usage for unique email check:
<SmartInput
  value={email}
  onChange={setEmail}
  validate={async (email) => {
    const res = await fetch(`/api/auth/check-email?email=${email}`);
    const { available } = await res.json();
    return {
      valid: available,
      message: available ? 'Email available' : 'Email already taken',
    };
  }}
/>
```

### 1.4 Undo/Redo Functionality

**Add Command Pattern for Undo/Redo:**

```typescript
// hooks/useUndoRedo.ts
import { useState, useCallback } from 'react';

interface Command<T> {
  execute: () => T;
  undo: () => T;
}

export function useUndoRedo<T>(initialState: T) {
  const [state, setState] = useState<T>(initialState);
  const [history, setHistory] = useState<Command<T>[]>([]);
  const [currentIndex, setCurrentIndex] = useState(-1);

  const executeCommand = useCallback((command: Command<T>) => {
    const newState = command.execute();
    setState(newState);

    // Clear any "future" history
    const newHistory = history.slice(0, currentIndex + 1);
    newHistory.push(command);
    setHistory(newHistory);
    setCurrentIndex(newHistory.length - 1);
  }, [history, currentIndex]);

  const undo = useCallback(() => {
    if (currentIndex < 0) return;

    const command = history[currentIndex];
    const newState = command.undo();
    setState(newState);
    setCurrentIndex(currentIndex - 1);
  }, [history, currentIndex]);

  const redo = useCallback(() => {
    if (currentIndex >= history.length - 1) return;

    const command = history[currentIndex + 1];
    const newState = command.execute();
    setState(newState);
    setCurrentIndex(currentIndex + 1);
  }, [history, currentIndex]);

  const canUndo = currentIndex >= 0;
  const canRedo = currentIndex < history.length - 1;

  return {
    state,
    executeCommand,
    undo,
    redo,
    canUndo,
    canRedo,
  };
}

// Usage in program builder:
const programBuilder = useUndoRedo(initialProgram);

// Add exercise command
const addExerciseCommand = {
  execute: () => ({
    ...programBuilder.state,
    exercises: [...programBuilder.state.exercises, newExercise],
  }),
  undo: () => ({
    ...programBuilder.state,
    exercises: programBuilder.state.exercises.slice(0, -1),
  }),
};

programBuilder.executeCommand(addExerciseCommand);

// UI
<div className="flex gap-2">
  <Button
    variant="outline"
    size="sm"
    onClick={programBuilder.undo}
    disabled={!programBuilder.canUndo}
  >
    <Undo2 className="h-4 w-4" />
    Undo (Ctrl+Z)
  </Button>
  <Button
    variant="outline"
    size="sm"
    onClick={programBuilder.redo}
    disabled={!programBuilder.canRedo}
```

```typescript
// hooks/useKeyboardShortcuts.ts
import { useEffect } from 'react';

interface ShortcutConfig {
  key: string;
  ctrlKey?: boolean;
  shiftKey?: boolean;
  altKey?: boolean;
  action: () => void;
  description: string;
}

export function useKeyboardShortcuts(shortcuts: ShortcutConfig[]) {
  useEffect(() => {
    function handleKeyDown(event: KeyboardEvent) {
      for (const shortcut of shortcuts) {
        const ctrlMatch = shortcut.ctrlKey ? event.ctrlKey || event.metaKey : !event.ctrlKey && !event.metaKey;
        const shiftMatch = shortcut.shiftKey ? event.shiftKey : !event.shiftKey;
        const altMatch = shortcut.altKey ? event.altKey : !event.altKey;
        const keyMatch = event.key.toLowerCase() === shortcut.key.toLowerCase();

        if (ctrlMatch && shiftMatch && altMatch && keyMatch) {
          event.preventDefault();
          shortcut.action();
          return;
        }
      }
    }

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [shortcuts]);
}
```

### 2.3 Focus Management

**Proper Focus Trapping in Modals:**

```typescript
// hooks/useFocusTrap.ts
import { useEffect, useRef } from 'react';

export function useFocusTrap(isActive: boolean) {
  const containerRef = useRef<HTMLDivElement>(null);
  const previousFocusRef = useRef<HTMLElement | null>(null);

  useEffect(() => {
    if (!isActive || !containerRef.current) return;

    // Store previous focus
    previousFocusRef.current = document.activeElement as HTMLElement;

    // Get focusable elements
    const focusableElements = containerRef.current.querySelectorAll(
      'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
    );

    const firstElement = focusableElements[0] as HTMLElement;
    const lastElement = focusableElements[focusableElements.length - 1] as HTMLElement;

    // Focus first element
    firstElement?.focus();

    function handleKeyDown(e: KeyboardEvent) {
      if (e.key !== 'Tab') return;

      if (e.shiftKey) {
        // Shift + Tab
        if (document.activeElement === firstElement) {
          e.preventDefault();
          lastElement?.focus();
        }
      } else {
        // Tab
        if (document.activeElement === lastElement) {
          e.preventDefault();
          firstElement?.focus();
        }
      }
    }

    document.addEventListener('keydown', handleKeyDown);

    return () => {
      document.removeEventListener('keydown', handleKeyDown);
      // Restore previous focus
      previousFocusRef.current?.focus();
    };
  }, [isActive]);

  return containerRef;
}

// Usage in Dialog:
export function AccessibleDialog({ open, children }: DialogProps) {
  const trapRef = useFocusTrap(open);

  return (
    <Dialog open={open}>
      <DialogContent ref={trapRef}>
        {children}
      </DialogContent>
    </Dialog>
  );
}
```

```typescript
// hooks/useHighContrast.ts
export function useHighContrast() {
  const [highContrast, setHighContrast] = useState(false);

  useEffect(() => {
    // Check if user prefers high contrast
    const mediaQuery = window.matchMedia('(prefers-contrast: high)');
    setHighContrast(mediaQuery.matches);

    const handler = (e: MediaQueryListEvent) => setHighContrast(e.matches);
    mediaQuery.addEventListener('change', handler);

    return () => mediaQuery.removeEventListener('change', handler);
  }, []);

  useEffect(() => {
    if (highContrast) {
      document.documentElement.classList.add('high-contrast');
    } else {
      document.documentElement.classList.remove('high-contrast');
    }
  }, [highContrast]);

  return { highContrast, toggleHighContrast: () => setHighContrast(!highContrast) };
}
```

### 3.2 Offline Workout Logging

**Add Offline Support with Background Sync:**

```typescript
// hooks/useOfflineWorkout.ts
import { useState, useEffect } from 'react';
import { useMutation } from '@tanstack/react-query';

interface OfflineWorkout {
  id: string;
  timestamp: number;
  data: any;
}

export function useOfflineWorkout() {
  const [isOnline, setIsOnline] = useState(navigator.onLine);
  const [pendingWorkouts, setPendingWorkouts] = useState<OfflineWorkout[]>([]);

  useEffect(() => {
    // Load pending workouts from localStorage
    const stored = localStorage.getItem('offline-workouts');
    if (stored) {
      setPendingWorkouts(JSON.parse(stored));
    }

    // Listen for online/offline events
    const handleOnline = () => setIsOnline(true);
    const handleOffline = () => setIsOnline(false);

    window.addEventListener('online', handleOnline);
    window.addEventListener('offline', handleOffline);

    return () => {
      window.removeEventListener('online', handleOnline);
      window.removeEventListener('offline', handleOffline);
    };
  }, []);

  const syncWorkouts = useMutation({
    mutationFn: async () => {
      const results = [];
      for (const workout of pendingWorkouts) {
        try {
          const res = await fetch('/api/workout-sessions', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(workout.data),
          });
          if (res.ok) {
            results.push({ id: workout.id, success: true });
          }
        } catch (error) {
          results.push({ id: workout.id, success: false });
        }
      }
      return results;
    },
    onSuccess: (results) => {
      // Remove successfully synced workouts
      const successIds = results
        .filter(r => r.success)
        .map(r => r.id);

      const remaining = pendingWorkouts.filter(
        w => !successIds.includes(w.id)
      );

      setPendingWorkouts(remaining);
      localStorage.setItem('offline-workouts', JSON.stringify(remaining));
    },
  });

  // Auto-sync when coming online
  useEffect(() => {
    if (isOnline && pendingWorkouts.length > 0) {
      syncWorkouts.mutate();
    }
  }, [isOnline, pendingWorkouts.length]);

  const saveOffline = (data: any) => {
    const workout: OfflineWorkout = {
      id: crypto.randomUUID(),
      timestamp: Date.now(),
      data,
    };

    const updated = [...pendingWorkouts, workout];
    setPendingWorkouts(updated);
    localStorage.setItem('offline-workouts', JSON.stringify(updated));
  };

  return {
    isOnline,
    pendingWorkouts,
    saveOffline,
    syncWorkouts,
  };
}

// Usage in workout page:
export function WorkoutPage() {
  const { isOnline, pendingWorkouts, saveOffline } = useOfflineWorkout();

  const logSet = async (data: any) => {
    if (!isOnline) {
      saveOffline(data);
      toast({
        title: 'Saved Offline',
        description: 'Will sync when connection is restored',
      });
      return;
    }

    // Normal API call
```

```typescript
export function useHaptic() {

    success: () => vibrate([10, 50, 10]),

}
```

```typescript
// server/routes/exports.routes.ts
import { Router } from 'express';
import { Parser } from 'json2csv';
import PDFDocument from 'pdfkit';

const router = Router();

router.get('/workouts', isAuthenticated, async (req: AuthRequest, res) => {
  const format = req.query.format || 'csv';
  const userId = req.currentUser!.id;

  // Fetch workout data
  const workouts = await workoutRepo.findByAthlete(userId);

  if (format === 'csv') {
    const fields = ['date', 'exercise', 'sets', 'reps', 'weight', 'rpe'];
    const parser = new Parser({ fields });
    const csv = parser.parse(workouts);

    res.setHeader('Content-Type', 'text/csv');
    res.setHeader('Content-Disposition', 'attachment; filename=workouts.csv');
    res.send(csv);
  } else if (format === 'json') {
    res.setHeader('Content-Type', 'application/json');
    res.setHeader('Content-Disposition', 'attachment; filename=workouts.json');
    res.json(workouts);
  } else if (format === 'pdf') {
    const doc = new PDFDocument();
    res.setHeader('Content-Type', 'application/pdf');
    res.setHeader('Content-Disposition', 'attachment; filename=workouts.pdf');

    doc.pipe(res);
    doc.fontSize(20).text('Workout History', { align: 'center' });
    doc.moveDown();

    // Add workout data to PDF
    workouts.forEach(workout => {
      doc.fontSize(14).text(`${workout.date} - ${workout.exercise}`);
      doc.fontSize(10).text(`${workout.sets}x${workout.reps} @ ${workout.weight}kg (RPE ${workout.rpe})`);
      doc.moveDown();
    });

    doc.end();
  }
});

export default router;
```

```typescript
router.get('/check-rest-day', isAuthenticated, async (req: AuthRequest, res) => {
  const userId = req.currentUser!.id;

  // Get last 10 days of workouts
  const tenDaysAgo = new Date();
  tenDaysAgo.setDate(tenDaysAgo.getDate() - 10);

  const workouts = await db.query.workoutSessions.findMany({
    where: and(
      eq(workoutSessions.athleteId, userId),
      gte(workoutSessions.date, tenDaysAgo)
    ),
    orderBy: desc(workoutSessions.date),
  });

  // Calculate consecutive days
  let consecutiveDays = 0;
  const today = new Date().setHours(0, 0, 0, 0);

  for (let i = 0; i < 10; i++) {
    const checkDate = new Date(today);
    checkDate.setDate(checkDate.getDate() - i);

    const hasWorkout = workouts.some((w) => {
      const workoutDate = new Date(w.date).setHours(0, 0, 0, 0);
      return workoutDate === checkDate.getTime();
    });

    if (hasWorkout) {
      consecutiveDays++;
    } else {
      break;
    }
  }

  res.json({
    needsRest: consecutiveDays >= 5,
    consecutiveDays,
  });
});
```

```typescript
// Brzycki formula for 1RM estimation
function calculateOneRM(weight: number, reps: number): number {
  if (reps === 1) return weight;
  return weight * (36 / (37 - reps));
}

// Server endpoint
router.get('/strength-progress', isAuthenticated, async (req: AuthRequest, res) => {
  const { athleteId, exerciseId } = req.query;

    .select({

      reps: sets.reps,

    .from(sets)
    .leftJoin(exerciseLogs, eq(exerciseLogs.id, sets.exerciseLogId))

      and(
        eq(workoutSessions.athleteId, athleteId as string),
        eq(exerciseLogs.exerciseId, exerciseId as string)
      )
    )
    .orderBy(workoutSessions.date);

  // Calculate estimated 1RM for each set
  const progress = sets.map((set) => ({
    date: set.date,
    estimatedOneRM: calculateOneRM(set.weight, set.reps),
    weight: set.weight,
    reps: set.reps,
  }));

  // Find PRs
  let maxOneRM = 0;
  const prs = progress.filter((entry) => {
    if (entry.estimatedOneRM > maxOneRM) {
      maxOneRM = entry.estimatedOneRM;
      return true;
    }
    return false;
  });

    dates: progress.map((p) => new Date(p.date).toLocaleDateString()),
    estimatedOneRMs: progress.map((p) => Math.round(p.estimatedOneRM)),
```

```typescript
export function VideoFeedback({ setId }: { setId: string }) {
  const [recording, setRecording] = useState(false);
  const videoRef = useRef<HTMLVideoElement>(null);
  const mediaRecorderRef = useRef<MediaRecorder | null>(null);

  const startRecording = async () => {
    const stream = await navigator.mediaDevices.getUserMedia({
      video: true,
      audio: true,
    });

    const mediaRecorder = new MediaRecorder(stream);
    mediaRecorderRef.current = mediaRecorder;

    const chunks: Blob[] = [];
    mediaRecorder.ondataavailable = (e) => chunks.push(e.data);

    mediaRecorder.onstop = async () => {
      const blob = new Blob(chunks, { type: 'video/webm' });
      await uploadFeedbackVideo(blob);
      stream.getTracks().forEach((track) => track.stop());
    };

    mediaRecorder.start();
    setRecording(true);
  };

  const stopRecording = () => {
    mediaRecorderRef.current?.stop();
    setRecording(false);
  };

  const uploadFeedbackVideo = async (blob: Blob) => {
    const formData = new FormData();
    formData.append('video', blob);
    formData.append('setId', setId);

    await fetch('/api/feedback/video', {
      method: 'POST',
      body: formData,
    });

    toast({ title: 'Feedback sent!' });
  };

  return (
    <Dialog>
      <DialogTrigger asChild>
        <Button variant="outline" size="sm">
```

```typescript
// lib/indexeddb.ts
import { openDB, DBSchema, IDBPDatabase } from 'idb';

interface WorkoutDB extends DBSchema {
  workouts: {
    key: string;
    value: {
      id: string;
      date: Date;
      exercises: any[];
      synced: boolean;
    };
  };
  exercises: {
    key: string;
    value: {
      id: string;
      name: string;
      category: string;
    };
  };
}

let db: IDBPDatabase<WorkoutDB>;

export async function initDB() {
  db = await openDB<WorkoutDB>('workout-app', 1, {
    upgrade(db) {
      db.createObjectStore('workouts', { keyPath: 'id' });
      db.createObjectStore('exercises', { keyPath: 'id' });
    },
  });
  return db;
}

export async function cacheWorkout(workout: any) {
  const db = await initDB();
  await db.put('workouts', { ...workout, synced: false });
}

export async function getUnsyncedWorkouts() {
  const db = await initDB();
  const allWorkouts = await db.getAll('workouts');
  return allWorkouts.filter((w) => !w.synced);
}

export async function markWorkoutSynced(id: string) {
  const db = await initDB();
  const workout = await db.get('workouts', id);
  if (workout) {
    await db.put('workouts', { ...workout, synced: true });
  }
}

// Hook to use IndexedDB
export function useOfflineWorkouts() {
  const syncWorkouts = async () => {
    const unsynced = await getUnsyncedWorkouts();

    for (const workout of unsynced) {
      try {
        await fetch('/api/workout-sessions', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(workout),
        });
        await markWorkoutSynced(workout.id);
      } catch (error) {
        console.error('Failed to sync workout', error);
      }
    }
  };

  useEffect(() => {
    // Sync when online
    if (navigator.onLine) {
      syncWorkouts();
    }

    window.addEventListener('online', syncWorkouts);
    return () => window.removeEventListener('online', syncWorkouts);
  }, []);
}
```

```typescript
export function requirePermission(permission: Permission) {
  return async (req: AuthRequest, res: Response, next: NextFunction) => {

      return res.status(403).json({
        error: 'Forbidden',
        message: 'You do not have permission to perform this action',
      });
    }

    next();
  };
}
```

```typescript
// Middleware to auto-log API calls
export function auditMiddleware(action: AuditAction) {
  return async (req: AuthRequest, res: Response, next: NextFunction) => {
    // Log after response
    const originalSend = res.send;
    res.send = function (data) {
      if (res.statusCode < 400) {

          path: req.path,
          method: req.method,
          ipAddress: req.ip,
          userAgent: req.get('user-agent'),

      return originalSend.call(this, data);

    next();
  };
```
