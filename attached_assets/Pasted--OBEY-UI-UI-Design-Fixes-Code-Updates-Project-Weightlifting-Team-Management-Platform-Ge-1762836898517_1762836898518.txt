### Fix 1.1: Create Reusable LoadingSpinner Component

**Problem:** Loading spinner code duplicated in 15+ places across the app.

```typescript
import React from 'react';

interface LoadingSpinnerProps {
  size?: 'sm' | 'md' | 'lg';
  message?: string;
  fullScreen?: boolean;
}

export function LoadingSpinner({
  size = 'md',
  message = 'Loading...',
  fullScreen = false
}: LoadingSpinnerProps) {
  const sizeClasses = {
    sm: 'h-4 w-4 border-2',
    md: 'h-8 w-8 border-4',
    lg: 'h-12 w-12 border-4',
  };

  const spinner = (
    <div className="text-center">
      <div
        className={`${sizeClasses[size]} mb-4 inline-block animate-spin rounded-full border-primary border-t-transparent`}
        role="status"
        aria-label="Loading"
      />
      {message && (
        <p className="text-muted-foreground" aria-live="polite">
          {message}
        </p>
      )}
    </div>
  );

  if (fullScreen) {
    return (
      <div className="flex h-screen items-center justify-center">
        {spinner}
      </div>
    );
  }

  return spinner;
}
```

```typescript
import React from 'react';
import { LucideIcon } from 'lucide-react';
import { Button } from '@/components/ui/button';

interface EmptyStateProps {
  icon: LucideIcon;
  title: string;
  description?: string;
  action?: {

  };

  icon: Icon,

          {action.icon && <action.icon className="mr-2 h-4 w-4" />}
          {action.label}
        </Button>
      )}
    </div>
  );

  if (variant === 'card') {
    return (
      <Card>
        <CardContent className="p-6">{content}</CardContent>
```

**Problem:** When API calls fail, users see nothing or infinite loading.

```typescript
import React from 'react';
import { AlertTriangle } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { Alert, AlertDescription, AlertTitle } from '@/components/ui/alert';

interface ErrorStateProps {
  title?: string;
  message?: string;
  onRetry?: () => void;
  variant?: 'inline' | 'card';
}

export function ErrorState({
  title = 'Something went wrong',
  message = 'An error occurred while loading data. Please try again.',
  onRetry,
  variant = 'card',
}: ErrorStateProps) {
  if (variant === 'inline') {
    return (
      <Alert variant="destructive">
        <AlertTriangle className="h-4 w-4" />
        <AlertTitle>{title}</AlertTitle>
        <AlertDescription className="mt-2">
          {message}
          {onRetry && (
            <Button
              variant="outline"
              size="sm"
              onClick={onRetry}
              className="mt-3"
            >
              Try Again
            </Button>
          )}
        </AlertDescription>
      </Alert>
    );
  }

  return (
    <div className="flex flex-col items-center justify-center py-12 text-center">
      <AlertTriangle className="mb-4 h-12 w-12 text-destructive" />
      <h3 className="mb-2 text-lg font-semibold">{title}</h3>
      <p className="mb-4 max-w-sm text-sm text-muted-foreground">{message}</p>
      {onRetry && (
        <Button onClick={onRetry} variant="outline">
          Try Again
        </Button>
      )}
    </div>
  );
}
```

```typescript
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { useToast } from '@/hooks/use-toast';

export function useProgramMutations() {
  const queryClient = useQueryClient();
  const { toast } = useToast();

  const createProgram = useMutation({
    mutationFn: async (data: { name: string; description?: string; organizationId: string }) => {
      const res = await fetch('/api/programs', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data),
      });
      if (!res.ok) throw new Error('Failed to create program');
      return res.json();
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['/api/programs'] });
      toast({ title: 'Program created successfully' });
    },
    onError: () => {
      toast({
        title: 'Failed to create program',
        description: 'Please try again',
        variant: 'destructive',
      });
    },
  });

  const addDay = useMutation({
    mutationFn: async ({
      weekId,
      dayNumber,
      name,
    }: {
      weekId: string;
      dayNumber: number;
      name?: string;
    }) => {
      const res = await fetch('/api/program-days', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ weekId, dayNumber, name }),
      });
      if (!res.ok) throw new Error('Failed to add day');
      return res.json();
    },
    onSuccess: (_, variables) => {
      queryClient.invalidateQueries({
        queryKey: ['/api/programs', variables.weekId, 'weeks'],
      });
      toast({ title: 'Day added successfully' });
    },
    onError: () => {
      toast({
        title: 'Failed to add day',
        variant: 'destructive',
      });
    },
  });

  return {
    createProgram,
    addDay,
  };
}
```

```typescript
import { useState, useEffect, useCallback } from 'react';

interface UseRestTimerOptions {
  onComplete?: () => void;
  autoStart?: boolean;
}

export function useRestTimer(options: UseRestTimerOptions = {}) {
  const { onComplete, autoStart = false } = options;
  const [isResting, setIsResting] = useState(autoStart);
  const [restTimer, setRestTimer] = useState(0);

  const startTimer = useCallback((seconds: number) => {
    setRestTimer(seconds);
    setIsResting(true);
  }, []);

  const stopTimer = useCallback(() => {
    setIsResting(false);
    setRestTimer(0);
  }, []);

  const pauseTimer = useCallback(() => {
    setIsResting(false);
  }, []);

  const resumeTimer = useCallback(() => {
    if (restTimer > 0) {
      setIsResting(true);
    }
  }, [restTimer]);

  useEffect(() => {
    if (!isResting || restTimer <= 0) return;

    const interval = setInterval(() => {
      setRestTimer((prev) => {
        const newValue = prev - 1;

        if (newValue <= 0) {
          setIsResting(false);
          onComplete?.();
          return 0;
        }

        return newValue;
      });
    }, 1000);

    return () => clearInterval(interval);
  }, [isResting, onComplete]); // Remove restTimer from deps to avoid re-creating interval

  const formatTime = useCallback((seconds: number) => {
    const mins = Math.floor(seconds / 60);
    const secs = seconds % 60;
    return `${mins}:${secs.toString().padStart(2, '0')}`;
  }, []);

  return {
    isResting,
    restTimer,
    formattedTime: formatTime(restTimer),
    startTimer,
    stopTimer,
    pauseTimer,
    resumeTimer,
  };
}
```

```typescript
  const num = typeof weight === 'string' ? parseFloat(weight) : weight;
```

```typescript
const addDay = useMutation({
  mutationFn: async ({ weekId, dayNumber, name }) => {
    const res = await fetch('/api/program-days', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ weekId, dayNumber, name }),
    });
    if (!res.ok) throw new Error('Failed to add day');
    return res.json();
  },
  onMutate: async (newDay) => {
    // Cancel outgoing refetches
    await queryClient.cancelQueries({
      queryKey: ['/api/programs', programId, 'weeks'],
    });

    // Snapshot previous value
    const previousWeeks = queryClient.getQueryData(['/api/programs', programId, 'weeks']);

    // Optimistically update
    queryClient.setQueryData(['/api/programs', programId, 'weeks'], (old: any) => {
      if (!old) return old;
      // Add new day optimistically
      return old.map((week: any) =>
        week.id === newDay.weekId
          ? {
              ...week,
              days: [...week.days, { ...newDay, id: 'temp-' + Date.now() }],
            }
          : week
      );
    });

    return { previousWeeks };
  },
  onError: (err, newDay, context) => {
    // Rollback on error
    queryClient.setQueryData(
      ['/api/programs', programId, 'weeks'],
      context?.previousWeeks
    );
    toast({
      title: 'Failed to add day',
      description: 'Your changes have been reverted',
      variant: 'destructive',
    });
  },
  onSettled: () => {
    // Refetch after error or success
    queryClient.invalidateQueries({
      queryKey: ['/api/programs', programId, 'weeks'],
    });
  },
```
