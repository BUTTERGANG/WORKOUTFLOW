```typescript
// PATCH program day name
app.patch('/api/program-days/:dayId', isAuthenticated, async (req: AuthRequest, res) => {
  try {
    const dayId = parseInt(req.params.dayId);
    const { name } = req.body;

    if (!name || typeof name !== 'string') {
      return res.status(400).json({ message: "Day name is required" });
    }

    // Get the day to verify access through program
    const day = await db.query.programDays.findFirst({
      where: (days, { eq }) => eq(days.id, dayId),
      with: {
        week: {
          with: {
            program: true
          }
        }
      }
    });

    if (!day) {
      return res.status(404).json({ message: "Program day not found" });
    }

    // Verify user has access to this program
    const userId = req.currentUser!.id;
    const program = day.week.program;

    if (program.createdBy !== userId) {
      // Check if user has access through organization
      const org = await db.query.organizations.findFirst({
        where: (orgs, { eq }) => eq(orgs.id, program.organizationId),
        with: {
          members: true
        }
      });

      const hasAccess = org?.members.some(m => m.userId === userId);
      if (!hasAccess && org?.ownerId !== userId) {
        return res.status(403).json({
          message: "You don't have permission to edit this program"
        });
      }
    }

    // Update the day name
    await db.update(programDays)
      .set({ name, updatedAt: new Date() })
      .where(eq(programDays.id, dayId));

    const updatedDay = await db.query.programDays.findFirst({
      where: (days, { eq }) => eq(days.id, dayId)
    });

    res.json(updatedDay);
  } catch (error: any) {
    logger.error("updating program day", error);
    res.status(500).json({ message: "Failed to update program day" });
  }
});
```

### Infrastructure #1: Health Check Endpoint

**Location:** Create new file `server/health.ts`

```typescript
import type { Express } from 'express';
import { db } from './db';
import { logger } from './logger';

export interface HealthStatus {
  status: 'healthy' | 'degraded' | 'unhealthy';
  timestamp: string;
  version: string;
  uptime: number;
  checks: {
    database: 'ok' | 'error';
    memory: 'ok' | 'warning' | 'critical';
    disk?: 'ok' | 'warning' | 'critical';
  };
  details?: Record<string, any>;
}

export function setupHealthChecks(app: Express) {
  // Basic health check (fast)
  app.get('/api/health', (req, res) => {
    res.json({
      status: 'healthy',
      timestamp: new Date().toISOString(),
      version: process.env.APP_VERSION || '1.0.0',
    });
  });

  // Detailed health check (includes DB check)
  app.get('/api/health/detailed', async (req, res) => {
    const startTime = Date.now();
    const health: HealthStatus = {
      status: 'healthy',
      timestamp: new Date().toISOString(),
      version: process.env.APP_VERSION || '1.0.0',
      uptime: process.uptime(),
      checks: {
        database: 'ok',
        memory: 'ok',
      },
      details: {},
    };

    // Check database
    try {
      await db.execute('SELECT 1');
      health.checks.database = 'ok';
    } catch (error) {
      logger.error('Health check: Database check failed', error);
      health.checks.database = 'error';
      health.status = 'unhealthy';
      health.details!.databaseError = error instanceof Error ? error.message : 'Unknown error';
    }

    // Check memory usage
    const memUsage = process.memoryUsage();
    const memoryUsagePercent = (memUsage.heapUsed / memUsage.heapTotal) * 100;

    if (memoryUsagePercent > 90) {
      health.checks.memory = 'critical';
      health.status = 'unhealthy';
    } else if (memoryUsagePercent > 75) {
      health.checks.memory = 'warning';
      if (health.status === 'healthy') health.status = 'degraded';
    }

    health.details!.memory = {
      heapUsed: Math.round(memUsage.heapUsed / 1024 / 1024) + ' MB',
      heapTotal: Math.round(memUsage.heapTotal / 1024 / 1024) + ' MB',
      usagePercent: Math.round(memoryUsagePercent) + '%',
    };

    health.details!.responseTime = (Date.now() - startTime) + 'ms';

    const statusCode = health.status === 'healthy' ? 200 :
                       health.status === 'degraded' ? 200 : 503;

    res.status(statusCode).json(health);
  });

  // Liveness probe (for Kubernetes/Docker)
  app.get('/api/health/live', (req, res) => {
    res.status(200).send('OK');
  });

  // Readiness probe (for Kubernetes/Docker)
  app.get('/api/health/ready', async (req, res) => {
    try {
      await db.execute('SELECT 1');
      res.status(200).send('OK');
    } catch (error) {
      res.status(503).send('Service Unavailable');
    }
  });
}
```

```typescript
import type { Request, Response, NextFunction } from 'express';
import { nanoid } from 'nanoid';

declare global {
  namespace Express {
    interface Request {
      requestId?: string;
    }
  }
}

export function requestIdMiddleware(req: Request, res: Response, next: NextFunction) {
  // Use existing request ID from header or generate new one
  const requestId = (req.headers['x-request-id'] as string) || nanoid();
  req.requestId = requestId;
  res.setHeader('X-Request-ID', requestId);
  next();
}
```

```typescript
let isShuttingDown = false;

// Graceful shutdown handler
async function gracefulShutdown(signal: string) {
  if (isShuttingDown) {
    logger.warn(`${signal} received again, forcing exit`);
    process.exit(1);
  }

  isShuttingDown = true;
  logger.info(`${signal} received, starting graceful shutdown`);

  // Stop accepting new connections
  server.close(async (err) => {
    if (err) {
      logger.error('Error during server shutdown', err);
      process.exit(1);
    }

    logger.info('HTTP server closed');

    try {
      // Close database connections
      await pool.end();
      logger.info('Database connections closed');

      logger.info('Graceful shutdown completed');
      process.exit(0);
    } catch (error) {
      logger.error('Error closing database connections', error);
      process.exit(1);
    }
  });

  // Force shutdown after 30 seconds
  setTimeout(() => {
    logger.error('Forced shutdown after timeout');
    process.exit(1);
  }, 30000);
}

// Handle shutdown signals
process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));
process.on('SIGINT', () => gracefulShutdown('SIGINT'));

// Handle uncaught errors
process.on('uncaughtException', (error) => {
  logger.error('Uncaught exception', error);
  gracefulShutdown('UNCAUGHT_EXCEPTION');
});

process.on('unhandledRejection', (reason, promise) => {
  logger.error('Unhandled rejection', { reason, promise });
  gracefulShutdown('UNHANDLED_REJECTION');
});
```

```typescript
import { z } from 'zod';

const envSchema = z.object({
  // Database
  DATABASE_URL: z.string().url().min(1, 'DATABASE_URL is required'),

  // Server
  NODE_ENV: z.enum(['development', 'production', 'test']).default('development'),
  PORT: z.string().regex(/^\d+$/).transform(Number).default('5000'),

  // Session & Security
  SESSION_SECRET: z.string().min(32, 'SESSION_SECRET must be at least 32 characters'),
  CSRF_SECRET: z.string().min(32).optional(),

  // Logging
  LOG_LEVEL: z.enum(['debug', 'info', 'warn', 'error']).optional(),

  // Application
  APP_VERSION: z.string().optional().default('1.0.0'),
  APP_URL: z.string().url().optional(),
});

export type Env = z.infer<typeof envSchema>;

let config: Env;

export function validateEnv(): Env {
  try {
    config = envSchema.parse(process.env);
    return config;
  } catch (error) {
    if (error instanceof z.ZodError) {
      console.error('❌ Environment variable validation failed:');
      error.errors.forEach((err) => {
        console.error(`  - ${err.path.join('.')}: ${err.message}`);
      });
      process.exit(1);
    }
    throw error;
  }
}

export function getConfig(): Env {
  if (!config) {
    throw new Error('Config not initialized. Call validateEnv() first.');
  }
  return config;
}
```

### Security #1: Enhanced Rate Limiting

**Location:** Update `server/middleware/rateLimit.ts`

```typescript
import type { Request, Response, NextFunction } from 'express';
import { logger } from '../logger';

interface RateLimitConfig {
  windowMs: number;      // Time window in milliseconds
  maxRequests: number;   // Max requests per window
  message?: string;
  skipSuccessfulRequests?: boolean;
  keyGenerator?: (req: Request) => string;
}

interface RateLimitEntry {
  count: number;
  resetTime: number;
  blockedUntil?: number;
}

const store = new Map<string, RateLimitEntry>();

// Clean up old entries every 5 minutes
setInterval(() => {
  const now = Date.now();
  for (const [key, entry] of store.entries()) {
    if (entry.resetTime < now && (!entry.blockedUntil || entry.blockedUntil < now)) {
      store.delete(key);
    }
  }
}, 5 * 60 * 1000);

export function rateLimit(config: RateLimitConfig) {
  const {
    windowMs,
    maxRequests,
    message = 'Too many requests, please try again later',
    skipSuccessfulRequests = false,
    keyGenerator = (req) => {
      // Use user ID if authenticated, otherwise IP
      const userId = (req as any).currentUser?.id;
      return userId ? `user:${userId}` : `ip:${req.ip || req.socket.remoteAddress}`;
    },
  } = config;

  return (req: Request, res: Response, next: NextFunction) => {
    const key = keyGenerator(req);
    const now = Date.now();

    let entry = store.get(key);

    // If entry doesn't exist or window has expired, create new entry
    if (!entry || entry.resetTime < now) {
      entry = {
        count: 0,
        resetTime: now + windowMs,
      };
      store.set(key, entry);
    }

    // Check if currently blocked
    if (entry.blockedUntil && entry.blockedUntil > now) {
      const retryAfter = Math.ceil((entry.blockedUntil - now) / 1000);
      res.setHeader('Retry-After', retryAfter.toString());
      res.setHeader('X-RateLimit-Limit', maxRequests.toString());
      res.setHeader('X-RateLimit-Remaining', '0');
      res.setHeader('X-RateLimit-Reset', new Date(entry.blockedUntil).toISOString());

      logger.warn('Rate limit exceeded', {
        key,
        path: req.path,
        blockedUntil: new Date(entry.blockedUntil).toISOString()
      });

      return res.status(429).json({
        message,
        retryAfter
      });
    }

    // Increment counter
    entry.count++;

    // Set rate limit headers
    const remaining = Math.max(0, maxRequests - entry.count);
    res.setHeader('X-RateLimit-Limit', maxRequests.toString());
    res.setHeader('X-RateLimit-Remaining', remaining.toString());
    res.setHeader('X-RateLimit-Reset', new Date(entry.resetTime).toISOString());

    // Check if limit exceeded
    if (entry.count > maxRequests) {
      // Block for 15 minutes on repeated violations
      entry.blockedUntil = now + (15 * 60 * 1000);

      logger.warn('Rate limit exceeded', {
        key,
        path: req.path,
        count: entry.count,
        maxRequests
      });

      const retryAfter = 900; // 15 minutes in seconds
      res.setHeader('Retry-After', retryAfter.toString());
      return res.status(429).json({
        message,
        retryAfter
      });
    }

    // If configured to skip successful requests, decrement on success
    if (skipSuccessfulRequests) {
      const originalSend = res.json.bind(res);
      res.json = function(body: any) {
        if (res.statusCode < 400) {
          entry.count = Math.max(0, entry.count - 1);
        }
        return originalSend(body);
      };
    }

    next();
  };
}

// Preset rate limiters
export const strictRateLimit = rateLimit({
  windowMs: 15 * 60 * 1000,  // 15 minutes
  maxRequests: 5,
  message: 'Too many attempts, please try again in 15 minutes',
});

export const authRateLimit = rateLimit({

  maxRequests: 10,
```

### Security #2: Input Sanitization Enhancement

**Location:** Update `server/middleware/sanitization.ts`

```typescript
import sanitizeHtml from 'sanitize-html';
import type { Request, Response, NextFunction } from 'express';

const MAX_STRING_LENGTH = 10000;
const MAX_ARRAY_LENGTH = 1000;
const MAX_OBJECT_DEPTH = 10;

// Define which fields should allow rich text
const RICH_TEXT_FIELDS = new Set([
  'description',
  'notes',
  'content',
  'bio',
  'message',
]);

// Define which fields should be stripped of ALL HTML
const PLAIN_TEXT_FIELDS = new Set([
  'name',
  'email',
  'username',
  'title',
  'firstName',
  'lastName',
]);

const ALLOWED_TAGS = [
  'p', 'br', 'strong', 'em', 'u', 's',
  'h1', 'h2', 'h3', 'h4', 'h5', 'h6',
  'ul', 'ol', 'li',
  'blockquote', 'code', 'pre',
  'a',
];

const ALLOWED_ATTRIBUTES = {
  'a': ['href', 'title', 'target'],
};

function sanitizeValue(
  value: any,
  fieldName: string,
  depth: number = 0
): any {
  if (depth > MAX_OBJECT_DEPTH) {
    throw new Error('Object nesting too deep');
  }

  if (value === null || value === undefined) {
    return value;
  }

  // Handle strings
  if (typeof value === 'string') {
    // Check length
    if (value.length > MAX_STRING_LENGTH) {
      throw new Error(`String too long in field: ${fieldName}`);
    }

    // Rich text fields
    if (RICH_TEXT_FIELDS.has(fieldName)) {
      return sanitizeHtml(value, {
        allowedTags: ALLOWED_TAGS,
        allowedAttributes: ALLOWED_ATTRIBUTES,
        allowedSchemes: ['http', 'https', 'mailto'],
      });
    }

    // Plain text fields - strip all HTML
    if (PLAIN_TEXT_FIELDS.has(fieldName)) {
      return sanitizeHtml(value, {
        allowedTags: [],
        allowedAttributes: {},
      });
    }

    // Default: allow basic formatting only
    return sanitizeHtml(value, {
      allowedTags: ['p', 'br', 'strong', 'em'],
      allowedAttributes: {},
    });
  }

  // Handle arrays
  if (Array.isArray(value)) {
    if (value.length > MAX_ARRAY_LENGTH) {
      throw new Error(`Array too large in field: ${fieldName}`);
    }
    return value.map((item, index) =>
      sanitizeValue(item, `${fieldName}[${index}]`, depth + 1)
    );
  }

  // Handle objects
  if (typeof value === 'object') {
    const sanitized: Record<string, any> = {};
    for (const [key, val] of Object.entries(value)) {
      sanitized[key] = sanitizeValue(val, key, depth + 1);
    }
    return sanitized;
  }

  // Numbers, booleans, etc. pass through
  return value;
}

export function sanitizeInputs(
  req: Request,
  res: Response,
  next: NextFunction
) {
  try {
    if (req.body) {
      req.body = sanitizeValue(req.body, 'body');
    }

    if (req.query) {
      const sanitized: Record<string, any> = {};
      for (const [key, value] of Object.entries(req.query)) {
        if (typeof value === 'string') {
          sanitized[key] = sanitizeValue(value, key);
        } else {
          sanitized[key] = value;
        }
      }
      req.query = sanitized;
    }

    next();
  } catch (error) {
    res.status(400).json({
      message: error instanceof Error ? error.message : 'Invalid input',
    });
  }
}
```

### Security #3: SQL Injection Prevention Check

**Location:** Create `server/middleware/sqlInjectionCheck.ts`

```typescript
import type { Request, Response, NextFunction } from 'express';
import { logger } from '../logger';

// Common SQL injection patterns
const SQL_INJECTION_PATTERNS = [
  /(\bunion\b.*\bselect\b)/i,
  /(\bselect\b.*\bfrom\b.*\bwhere\b)/i,
  /(\binsert\b.*\binto\b.*\bvalues\b)/i,
  /(\bdelete\b.*\bfrom\b)/i,
  /(\bdrop\b.*\btable\b)/i,
  /(\bexec\b.*\()/i,
  /(\bexecute\b.*\()/i,
  /(;.*\b(drop|delete|update|insert)\b)/i,
  /('.*or.*'.*=.*')/i,
  /(\-\-)/,
  /(\/\*.*\*\/)/,
  /(\bxp_.*\b)/i,
];

function checkForSqlInjection(value: any): boolean {
  if (typeof value === 'string') {
    return SQL_INJECTION_PATTERNS.some(pattern => pattern.test(value));
  }

  if (Array.isArray(value)) {
    return value.some(item => checkForSqlInjection(item));
  }

  if (typeof value === 'object' && value !== null) {
    return Object.values(value).some(val => checkForSqlInjection(val));
  }

  return false;
}

export function sqlInjectionCheck(
  req: Request,
  res: Response,
  next: NextFunction
) {
  // Check body
  if (req.body && checkForSqlInjection(req.body)) {
    logger.warn('Potential SQL injection attempt detected', {
      path: req.path,
      method: req.method,
      ip: req.ip,
      body: req.body,
    });

    return res.status(400).json({
      message: 'Invalid input detected',
    });
  }

  // Check query params
  if (req.query && checkForSqlInjection(req.query)) {
    logger.warn('Potential SQL injection attempt in query params', {
      path: req.path,
      method: req.method,
      ip: req.ip,
      query: req.query,
    });

    return res.status(400).json({
      message: 'Invalid query parameters',
    });
  }

  // Check URL params
  if (req.params && checkForSqlInjection(req.params)) {
    logger.warn('Potential SQL injection attempt in URL params', {
      path: req.path,
      method: req.method,
      ip: req.ip,
      params: req.params,
    });

    return res.status(400).json({
      message: 'Invalid URL parameters',
    });
  }

  next();
}
```

```typescript
interface PaginationOptions {
  page?: number;
  limit?: number;
}

interface PaginatedResult<T> {
  data: T[];
  pagination: {
    page: number;
    limit: number;
    total: number;
    totalPages: number;
    hasNext: boolean;
    hasPrev: boolean;
  };
}

export async function paginate<T>(
  query: any,
  countQuery: any,
  options: PaginationOptions = {}
): Promise<PaginatedResult<T>> {
  const page = Math.max(1, options.page || 1);
  const limit = Math.min(100, Math.max(1, options.limit || 20));
  const offset = (page - 1) * limit;

  const [data, countResult] = await Promise.all([
    query.limit(limit).offset(offset),
    countQuery,
  ]);

  const total = countResult[0]?.count || 0;
  const totalPages = Math.ceil(total / limit);

  return {
    data,
    pagination: {
      page,
      limit,
      total,
      totalPages,
      hasNext: page < totalPages,
      hasPrev: page > 1,
    },
  };
}
```

### Performance #3: Response Caching

**Location:** Create `server/middleware/cache.ts`

```typescript
import type { Request, Response, NextFunction } from 'express';

interface CacheOptions {
  ttl: number; // Time to live in seconds
  key?: (req: Request) => string;
}

const cache = new Map<string, { data: any; expires: number }>();

// Clean up expired cache entries every minute
setInterval(() => {
  const now = Date.now();
  for (const [key, value] of cache.entries()) {
    if (value.expires < now) {
      cache.delete(key);
    }
  }
}, 60 * 1000);

export function cacheMiddleware(options: CacheOptions) {
  const { ttl, key: keyGenerator } = options;

  return (req: Request, res: Response, next: NextFunction) => {
    // Only cache GET requests
    if (req.method !== 'GET') {
      return next();
    }

    // Generate cache key
    const cacheKey = keyGenerator
      ? keyGenerator(req)
      : `${req.path}:${JSON.stringify(req.query)}`;

    // Check cache
    const cached = cache.get(cacheKey);
    if (cached && cached.expires > Date.now()) {
      res.setHeader('X-Cache', 'HIT');
      res.setHeader('Cache-Control', `public, max-age=${ttl}`);
      return res.json(cached.data);
    }

    // Intercept response
    const originalJson = res.json.bind(res);
    res.json = function(data: any) {
      // Only cache successful responses
      if (res.statusCode === 200) {
        cache.set(cacheKey, {
          data,
          expires: Date.now() + (ttl * 1000),
        });
      }

      res.setHeader('X-Cache', 'MISS');
      res.setHeader('Cache-Control', `public, max-age=${ttl}`);
      return originalJson(data);
    };

    next();
  };
}

export function clearCache(pattern?: string | RegExp) {
  if (!pattern) {
    cache.clear();
    return;
  }

  const regex = typeof pattern === 'string'
    ? new RegExp(pattern)
    : pattern;

  for (const key of cache.keys()) {
    if (regex.test(key)) {
      cache.delete(key);
    }
  }
}
```

```typescript
import * as Sentry from '@sentry/node';
import type { Express } from 'express';

export function initSentry(app: Express) {
  if (!process.env.SENTRY_DSN) {
    console.warn('SENTRY_DSN not configured, error tracking disabled');
    return;
  }

  Sentry.init({
    dsn: process.env.SENTRY_DSN,
    environment: process.env.NODE_ENV || 'development',
    tracesSampleRate: process.env.NODE_ENV === 'production' ? 0.1 : 1.0,

    // Performance monitoring
    integrations: [
      // HTTP integration for tracing
      new Sentry.Integrations.Http({ tracing: true }),
      // Express integration
      new Sentry.Integrations.Express({ app }),
    ],

    // Filter out sensitive data
    beforeSend(event, hint) {
      // Remove sensitive headers
      if (event.request?.headers) {
        delete event.request.headers['authorization'];
        delete event.request.headers['cookie'];
      }

      // Remove sensitive body data
      if (event.request?.data) {
        const data = event.request.data;
        if (typeof data === 'object') {
          delete data.password;
          delete data.passwordHash;
          delete data.token;
        }
      }

      return event;
    },
  });

  // Request handler must be the first middleware
  app.use(Sentry.Handlers.requestHandler());

  // Tracing handler
  app.use(Sentry.Handlers.tracingHandler());
}

export function setupSentryErrorHandler(app: Express) {
  // Error handler must be before any other error middleware
  app.use(Sentry.Handlers.errorHandler());
}
```

### Monitoring #2: Performance Metrics

**Create `server/monitoring/metrics.ts`:**

```typescript
import type { Request, Response, NextFunction } from 'express';
import { logger } from '../logger';

interface Metrics {
  requests: {
    total: number;
    by_status: Record<string, number>;
    by_method: Record<string, number>;
    by_route: Record<string, number>;
  };
  response_times: {
    count: number;
    sum: number;
    avg: number;
    min: number;
    max: number;
    p50: number;
    p95: number;
    p99: number;
  };
  errors: {
    total: number;
    by_type: Record<string, number>;
  };
}

const metrics: Metrics = {
  requests: {
    total: 0,
    by_status: {},
    by_method: {},
    by_route: {},
  },
  response_times: {
    count: 0,
    sum: 0,
    avg: 0,
    min: Infinity,
    max: 0,
    p50: 0,
    p95: 0,
    p99: 0,
  },
  errors: {
    total: 0,
    by_type: {},
  },
};

const responseTimes: number[] = [];
const MAX_RESPONSE_TIMES = 1000; // Keep last 1000 response times

export function metricsMiddleware(
  req: Request,
  res: Response,
  next: NextFunction
) {
  const startTime = Date.now();

  // Capture response finish
  res.on('finish', () => {
    const duration = Date.now() - startTime;

    // Update metrics
    metrics.requests.total++;

    // By status
    const status = res.statusCode.toString();
    metrics.requests.by_status[status] = (metrics.requests.by_status[status] || 0) + 1;

    // By method
    metrics.requests.by_method[req.method] = (metrics.requests.by_method[req.method] || 0) + 1;

    // By route (remove IDs for aggregation)
    const route = req.path.replace(/\/\d+/g, '/:id');
    metrics.requests.by_route[route] = (metrics.requests.by_route[route] || 0) + 1;

    // Response times
    responseTimes.push(duration);
    if (responseTimes.length > MAX_RESPONSE_TIMES) {
      responseTimes.shift();
    }

    metrics.response_times.count++;
    metrics.response_times.sum += duration;
    metrics.response_times.avg = metrics.response_times.sum / metrics.response_times.count;
    metrics.response_times.min = Math.min(metrics.response_times.min, duration);
    metrics.response_times.max = Math.max(metrics.response_times.max, duration);

    // Calculate percentiles
    const sorted = [...responseTimes].sort((a, b) => a - b);
    metrics.response_times.p50 = sorted[Math.floor(sorted.length * 0.5)] || 0;
    metrics.response_times.p95 = sorted[Math.floor(sorted.length * 0.95)] || 0;
    metrics.response_times.p99 = sorted[Math.floor(sorted.length * 0.99)] || 0;

    // Log slow requests
    if (duration > 1000) {
      logger.warn('Slow request detected', {
        method: req.method,
        path: req.path,
        duration: `${duration}ms`,
        status: res.statusCode,
      });
    }
  });

  next();
}

export function trackError(errorType: string) {
  metrics.errors.total++;
  metrics.errors.by_type[errorType] = (metrics.errors.by_type[errorType] || 0) + 1;
}

export function getMetrics(): Metrics {
  return JSON.parse(JSON.stringify(metrics)); // Deep clone
}

export function resetMetrics() {
  metrics.requests.total = 0;
  metrics.requests.by_status = {};
  metrics.requests.by_method = {};
  metrics.requests.by_route = {};
  metrics.response_times.count = 0;
  metrics.response_times.sum = 0;
  metrics.response_times.avg = 0;
  metrics.response_times.min = Infinity;
  metrics.response_times.max = 0;
  metrics.errors.total = 0;
  metrics.errors.by_type = {};
  responseTimes.length = 0;
}
```

```typescript
import { useState, useEffect } from 'react';
import { useToast } from './use-toast';

export function useOnlineStatus() {
  const [isOnline, setIsOnline] = useState(navigator.onLine);
  const { toast } = useToast();

  useEffect(() => {
    function handleOnline() {
      setIsOnline(true);
      toast({
        title: 'Back online',
        description: 'Connection restored',
      });
    }

    function handleOffline() {
      setIsOnline(false);
      toast({
        title: 'Connection lost',
        description: 'Some features may not work offline',
        variant: 'destructive',
      });
    }

    window.addEventListener('online', handleOnline);
    window.addEventListener('offline', handleOffline);

    return () => {
      window.removeEventListener('online', handleOnline);
      window.removeEventListener('offline', handleOffline);
    };
  }, [toast]);

  return isOnline;
}
```

### Beta #2: Sample Data Generator

**Create `server/seedSampleData.ts`:**

```typescript
import { db } from './db';
import {
  users,
  organizations,
  teams,
  programs,
  programWeeks,
  programDays,
  programExercises,
  exercises,
} from '../shared/schema';
import { hashPassword } from './localAuth';

export async function seedSampleData(userId: number) {
  try {
    // Create sample exercises
    const sampleExercises = [
      { name: 'Back Squat', category: 'squat', muscleGroup: 'legs', equipment: 'barbell' },
      { name: 'Bench Press', category: 'press', muscleGroup: 'chest', equipment: 'barbell' },
      { name: 'Deadlift', category: 'deadlift', muscleGroup: 'back', equipment: 'barbell' },
      { name: 'Front Squat', category: 'squat', muscleGroup: 'legs', equipment: 'barbell' },
      { name: 'Clean', category: 'clean', muscleGroup: 'full-body', equipment: 'barbell' },
    ];

    const createdExercises = await db.insert(exercises)
      .values(sampleExercises)
      .returning();

    // Create sample program
    const org = await db.query.organizations.findFirst({
      where: (orgs, { eq }) => eq(orgs.ownerId, userId),
    });

    if (!org) {
      throw new Error('User must have an organization first');
    }

    const program = await db.insert(programs).values({
      name: 'Sample Strength Program',
      description: 'A sample 4-week strength program to get you started',
      durationWeeks: 4,
      phase: 'strength',
      organizationId: org.id,
      createdBy: userId,
    }).returning();

    // Create weeks
    for (let weekNum = 1; weekNum <= 4; weekNum++) {
      const week = await db.insert(programWeeks).values({
        programId: program[0].id,
        weekNumber: weekNum,
        name: `Week ${weekNum}`,
      }).returning();

      // Create 3 days per week
      for (let dayNum = 1; dayNum <= 3; dayNum++) {
        const day = await db.insert(programDays).values({
          weekId: week[0].id,
          dayNumber: dayNum,
          name: ['Squat Day', 'Bench Day', 'Deadlift Day'][dayNum - 1],
        }).returning();

        // Add exercises to day
        const dayExercises = [
          [createdExercises[0], createdExercises[3]], // Squat Day
          [createdExercises[1]], // Bench Day
          [createdExercises[2]], // Deadlift Day
        ][dayNum - 1];

        for (let i = 0; i < dayExercises.length; i++) {
          await db.insert(programExercises).values({
            dayId: day[0].id,
            exerciseId: dayExercises[i].id,
            sets: 5,
            reps: 5,
            intensity: 80,
            rest: 180,
            order: i + 1,
          });
        }
      }
    }

    return { success: true, programId: program[0].id };
  } catch (error) {
    console.error('Error seeding sample data:', error);
    throw error;
  }
}
```
