## Executive Summary

### Current Architecture Analysis

**Code Statistics:**
- `server/routes.ts`: 2,341 lines, 72 endpoints
- `server/storage.ts`: 1,744 lines, 50+ methods
- `client/src/pages/programs.tsx`: 1,323 lines
- Database tables: 21 tables
- Total API endpoints: 72

**Architecture Pattern:**
```
Current: Route Handler → Storage (Repository) → Database
```

**Strengths:**
- ✅ Solid security foundation (CSRF, rate limiting, sanitization)
- ✅ Well-designed database schema with proper relationships
- ✅ Type-safe with TypeScript throughout
- ✅ Comprehensive error handling and logging
- ✅ Good validation with Zod schemas

**Growth Challenges:**
- ⚠️ Monolithic files becoming hard to navigate
- ⚠️ Business logic mixed with route handlers
- ⚠️ Repeated authorization patterns
- ⚠️ Large React components with multiple responsibilities
- ⚠️ No clear separation between layers

## 🔴 HIGH IMPACT REDESIGNS

### 1. Modularize Route Handlers (2,341 lines → 7 modules)

**Priority:** P0 - Should do before adding more features
**Effort:** Medium (2-3 days)
**Impact:** High - Immediate improvement to maintainability

#### Current Structure

```
server/
  routes.ts (2,341 lines containing ALL endpoints)
```

**Problems:**
- Difficult to navigate and find specific endpoints
- Merge conflicts when multiple developers work on routes
- Hard to test individual domains in isolation
- Business logic mixed with routing concerns
- No clear boundaries between features

#### Proposed Structure

```
server/
  routes/
    index.ts                    # Route registry
    auth.routes.ts              # Authentication (register, login, logout)
    organizations.routes.ts     # Organization CRUD + membership
    teams.routes.ts             # Team management + join requests
    programs.routes.ts          # Program builder (create, weeks, days)
    workouts.routes.ts          # Workout logging + sessions
    messages.routes.ts          # Messaging system
    exercises.routes.ts         # Exercise library
    statistics.routes.ts        # Analytics and stats
```

**`server/routes/auth.routes.ts`** (Example Module):

```typescript
import type { Express } from 'express';
import { z } from 'zod';
import { storage } from '../storage';
import { logger } from '../logger';
import { setupLocalAuth, isAuthenticated, hashPassword, type AuthRequest } from '../localAuth';
import { rateLimit } from '../middleware/rateLimit';

const registerSchema = z.object({
  email: z.string().email(),
  password: z.string().min(12, "Password must be at least 12 characters"),
  firstName: z.string().min(2),
  lastName: z.string().min(2),
  role: z.enum(['admin', 'head_coach', 'assistant_coach', 'athlete']),
});

const loginSchema = z.object({
  email: z.string().email(),
  password: z.string(),
});

export function registerAuthRoutes(app: Express) {
  // Setup authentication middleware
  setupLocalAuth(app);

  // Get current user
  app.get('/api/auth/user', isAuthenticated, async (req: AuthRequest, res) => {
    try {
      if (!req.currentUser) {
        return res.status(401).json({ message: "Unauthorized" });
      }

      const { passwordHash, ...userWithoutPassword } = req.currentUser;
      res.json(userWithoutPassword);
    } catch (error) {
      logger.error("fetching user", error);
      res.status(500).json({ message: "Failed to fetch user" });
    }
  });

  // Register new user
  app.post('/api/auth/register', rateLimit(5, 15 * 60 * 1000), async (req, res) => {
    try {
      const data = registerSchema.parse(req.body);

      const existingUser = await storage.getUserByEmail(data.email);
      if (existingUser) {
        return res.status(400).json({ message: "Email already registered" });
      }

      const passwordHash = await hashPassword(data.password);
      const user = await storage.createUser({
        email: data.email,
        passwordHash,
        firstName: data.firstName,
        lastName: data.lastName,
        role: data.role,
      });

      req.login(user, (err) => {
        if (err) {
          logger.error("logging in user after registration", err);
          return res.status(500).json({
            message: "Registration successful but login failed"
          });
        }

        const { passwordHash, ...userWithoutPassword } = user;
        res.json(userWithoutPassword);
      });
    } catch (error: any) {
      logger.error("registering user", error);
      if (error.name === 'ZodError') {
        return res.status(400).json({
          message: "Validation error",
          errors: error.errors
        });
      }
      res.status(400).json({ message: "Failed to register user" });
    }
  });

  // Login
  app.post('/api/auth/login', rateLimit(5, 15 * 60 * 1000), async (req, res, next) => {
    try {
      const data = loginSchema.parse(req.body);

      // Passport authentication handled by middleware
      passport.authenticate('local', (err: any, user: any, info: any) => {
        if (err) {
          logger.error("login error", err);
          return res.status(500).json({ message: "Login failed" });
        }

        if (!user) {
          return res.status(401).json({
            message: info?.message || "Invalid credentials"
          });
        }

        req.login(user, (err) => {
          if (err) {
            logger.error("session creation failed", err);
            return res.status(500).json({ message: "Login failed" });
          }

          const { passwordHash, ...userWithoutPassword } = user;
          res.json(userWithoutPassword);
        });
      })(req, res, next);
    } catch (error: any) {
      logger.error("login validation error", error);
      res.status(400).json({ message: "Invalid request" });
    }
  });

  // Logout
  app.post('/api/auth/logout', (req, res) => {
    req.logout((err) => {
      if (err) {
        logger.error("logout error", err);
        return res.status(500).json({ message: "Logout failed" });
      }
      res.json({ message: "Logged out successfully" });
    });
  });
}
```

**`server/routes/programs.routes.ts`** (Example Module):

```typescript
import type { Express } from 'express';
import { z } from 'zod';
import { storage } from '../storage';
import { logger } from '../logger';
import { isAuthenticated, type AuthRequest } from '../localAuth';
import { verifyOrganizationAccess, hasOrganizationAccess } from '../middleware/authorization';
import {
  insertProgramSchema,
  insertProgramWeekSchema,
  insertProgramDaySchema,
  insertProgramExerciseSchema,
  insertProgramAssignmentSchema,
} from '@shared/schema';

export function registerProgramRoutes(app: Express) {

  // Create program
  app.post('/api/programs', isAuthenticated, async (req: AuthRequest, res) => {
    try {
      const userId = req.currentUser!.id;
      const data = insertProgramSchema.parse({ ...req.body, createdBy: userId });

      // Verify user has access to the organization
      const hasAccess = await hasOrganizationAccess(userId, data.organizationId);
      if (!hasAccess) {
        return res.status(403).json({
          message: "You don't have access to this organization"
        });
      }

      const program = await storage.createProgram(data);
      res.json(program);
    } catch (error: any) {
      logger.error("creating program", error);
      res.status(400).json({ message: "Failed to create program" });
    }
  });

  // Get programs for organization
  app.get('/api/programs', isAuthenticated, async (req: AuthRequest, res) => {
    try {
      const { organizationId } = req.query;

      if (!organizationId) {
        return res.status(400).json({
          message: "organizationId is required"
        });
      }

      const hasAccess = await hasOrganizationAccess(
        req.currentUser!.id,
        organizationId as string
      );

      if (!hasAccess) {
        return res.status(403).json({
          message: "You don't have access to this organization"
        });
      }

      const programs = await storage.getOrganizationPrograms(organizationId as string);
      res.json(programs);
    } catch (error: any) {
      logger.error("fetching programs", error);
      res.status(500).json({ message: "Failed to fetch programs" });
    }
  });

  // Get program by ID (with full structure)
  app.get('/api/programs/:id', isAuthenticated, async (req: AuthRequest, res) => {
    try {
      const program = await storage.getProgram(req.params.id);

      if (!program) {
        return res.status(404).json({ message: "Program not found" });
      }

      const hasAccess = await hasOrganizationAccess(
        req.currentUser!.id,
        program.organizationId
      );

      if (!hasAccess) {
        return res.status(403).json({ message: "Forbidden" });
      }

      res.json(program);
    } catch (error: any) {
      logger.error("fetching program", error);
      res.status(500).json({ message: "Failed to fetch program" });
    }
  });

  // Delete program
  app.delete('/api/programs/:id', isAuthenticated, async (req: AuthRequest, res) => {
    try {
      const program = await storage.getProgram(req.params.id);

      if (!program) {
        return res.status(404).json({ message: "Program not found" });
      }

      // Verify user created the program or is org owner
      if (program.createdBy !== req.currentUser!.id) {
        const org = await storage.getOrganization(program.organizationId);
        if (org?.ownerId !== req.currentUser!.id) {
          return res.status(403).json({
            message: "Only program creator or org owner can delete"
          });
        }
      }

      await storage.deleteProgram(req.params.id);
      res.json({ message: "Program deleted successfully" });
    } catch (error: any) {
      logger.error("deleting program", error);
      res.status(500).json({ message: "Failed to delete program" });
    }
  });

  // Create program week
  app.post('/api/programs/:programId/weeks', isAuthenticated, async (req: AuthRequest, res) => {
    try {
      const program = await storage.getProgram(req.params.programId);

      if (!program) {
        return res.status(404).json({ message: "Program not found" });
      }

      const hasAccess = await hasOrganizationAccess(
        req.currentUser!.id,
        program.organizationId
      );

      if (!hasAccess) {
        return res.status(403).json({ message: "Forbidden" });
      }

      const data = insertProgramWeekSchema.parse({
        ...req.body,
        programId: req.params.programId,
      });

      const week = await storage.createProgramWeek(data);
      res.json(week);
    } catch (error: any) {
      logger.error("creating program week", error);
      res.status(400).json({ message: "Failed to create program week" });
    }
  });

  // Get program weeks
  app.get('/api/programs/:programId/weeks', isAuthenticated, async (req: AuthRequest, res) => {
    try {
      const program = await storage.getProgram(req.params.programId);

      if (!program) {
        return res.status(404).json({ message: "Program not found" });
      }

      const hasAccess = await hasOrganizationAccess(
        req.currentUser!.id,
        program.organizationId
      );

      if (!hasAccess) {
        return res.status(403).json({ message: "Forbidden" });
      }

      const weeks = await storage.getProgramWeeks(req.params.programId);
      res.json(weeks);
    } catch (error: any) {
      logger.error("fetching program weeks", error);
      res.status(500).json({ message: "Failed to fetch program weeks" });
    }
  });

  // Create program day
  app.post('/api/weeks/:weekId/days', isAuthenticated, async (req: AuthRequest, res) => {
    try {
      const week = await storage.getProgramWeek(req.params.weekId);

      if (!week) {
        return res.status(404).json({ message: "Week not found" });
      }

      const program = await storage.getProgram(week.programId);
      if (!program) {
        return res.status(404).json({ message: "Program not found" });
      }

      const hasAccess = await hasOrganizationAccess(
        req.currentUser!.id,
        program.organizationId
      );

      if (!hasAccess) {
        return res.status(403).json({ message: "Forbidden" });
      }

      const data = insertProgramDaySchema.parse({
        ...req.body,
        weekId: req.params.weekId,
      });

      const day = await storage.createProgramDay(data);
      res.json(day);
    } catch (error: any) {
      logger.error("creating program day", error);
      res.status(400).json({ message: "Failed to create program day" });
    }
  });

  // Additional program routes...
  // - PATCH /api/programs/:id (update)
  // - GET /api/weeks/:weekId/days (get days)
  // - POST /api/program-days/:dayId/exercises (add exercise)
  // - etc.
}
```

### 2. Introduce Service Layer (Business Logic Separation)

**Priority:** P0 - Critical for maintainability
**Effort:** High (1-2 weeks)
**Impact:** High - Cleaner code, better testability

#### Current Architecture Problems

**Example from routes.ts (lines 1000-1045):**

```typescript
app.post('/api/weeks/:weekId/days', async (req, res) => {
  // 1. Get week from database
  const week = await storage.getProgramWeek(req.params.weekId);
  if (!week) {
    return res.status(404).json({ message: "Week not found" });
  }

  // 2. Get program from database
  const program = await storage.getProgram(week.programId);
  if (!program) {
    return res.status(404).json({ message: "Program not found" });
  }

  // 3. Get organization from database
  const org = await storage.getOrganization(program.organizationId);
  if (!org) {
    return res.status(404).json({ message: "Organization not found" });
  }

  // 4. Complex authorization logic
  const isOwner = org.ownerId === req.currentUser!.id;
  const isCoach = req.currentUser!.role === 'admin' ||
                 req.currentUser!.role === 'head_coach' ||
                 req.currentUser!.role === 'assistant_coach';

  if (!isOwner && !isCoach) {
    return res.status(403).json({
      message: "Forbidden: only coaches or org owners can modify programs"
    });
  }

  // 5. Additional access verification
  const hasAccess = await hasOrganizationAccess(
    req.currentUser!.id,
    program.organizationId
  );

  if (!hasAccess) {
    return res.status(403).json({
      message: "Forbidden: program not in your organization"
    });
  }

  // 6. Finally, create the day
  const data = insertProgramDaySchema.parse({
    ...req.body,
    weekId: req.params.weekId
  });

  const day = await storage.createProgramDay(data);
  res.json(day);
});
```

**Problems:**
- 45 lines of route handler code
- Business logic mixed with HTTP concerns
- Authorization logic repeated across routes
- Hard to test without mocking HTTP request/response
- Can't reuse logic for scheduled jobs, CLI tools, etc.

#### Proposed Architecture

```
┌─────────────┐
│   Route     │  HTTP concerns only
│  Handler    │  - Parse request
└──────┬──────┘  - Call service
       │         - Format response
       ▼
┌─────────────┐
│  Service    │  Business logic
│   Layer     │  - Validation
└──────┬──────┘  - Authorization
       │         - Orchestration
       ▼
┌─────────────┐
│ Repository  │  Data access
│   Layer     │  - Database queries
└─────────────┘  - Transactions
```

#### Implementation Example

**`server/services/program.service.ts`**:

```typescript
import { ProgramRepository } from '../repositories/program.repository';
import { OrganizationRepository } from '../repositories/organization.repository';
import { AuthorizationService } from './authorization.service';
import { NotFoundError, ForbiddenError, ValidationError } from '../errors';
import { logger } from '../logger';
import type {
  CreateProgramDto,
  CreateProgramWeekDto,
  CreateProgramDayDto,
  Program,
  ProgramWeek,
  ProgramDay,
} from '../types';

export class ProgramService {
  constructor(
    private programRepo: ProgramRepository,
    private orgRepo: OrganizationRepository,
    private authService: AuthorizationService,
  ) {}

  async createProgram(data: CreateProgramDto, userId: string): Promise<Program> {
    // Verify user has access to organization
    await this.authService.verifyOrganizationMembership(userId, data.organizationId);

    // Business rule: Only coaches can create programs
    const user = await this.authService.getUser(userId);
    this.authService.verifyCoachRole(user);

    // Create the program
    const program = await this.programRepo.create({
      ...data,
      createdBy: userId,
    });

    logger.info('Program created', {
      programId: program.id,
      userId,
      organizationId: data.organizationId,
    });

    return program;
  }

  async getProgram(programId: string, userId: string): Promise<Program> {
    const program = await this.programRepo.findById(programId);

    if (!program) {
      throw new NotFoundError('Program not found');
    }

    // Verify user has access to this program's organization
    await this.authService.verifyOrganizationMembership(
      userId,
      program.organizationId
    );

    return program;
  }

  async deleteProgram(programId: string, userId: string): Promise<void> {
    const program = await this.programRepo.findById(programId);

    if (!program) {
      throw new NotFoundError('Program not found');
    }

    // Business rule: Only creator or org owner can delete
    const isCreator = program.createdBy === userId;
    const org = await this.orgRepo.findById(program.organizationId);
    const isOrgOwner = org?.ownerId === userId;

    if (!isCreator && !isOrgOwner) {
      throw new ForbiddenError(
        'Only program creator or organization owner can delete programs'
      );
    }

    await this.programRepo.delete(programId);

    logger.info('Program deleted', { programId, userId });
  }

  async addWeekToProgram(
    programId: string,
    data: CreateProgramWeekDto,
    userId: string
  ): Promise<ProgramWeek> {
    // Verify program exists and user has access
    const program = await this.getProgram(programId, userId);

    // Business rule: Validate week number sequence
    const existingWeeks = await this.programRepo.getWeeks(programId);
    const expectedWeekNumber = existingWeeks.length + 1;

    if (data.weekNumber !== expectedWeekNumber) {
      throw new ValidationError(
        `Week number must be ${expectedWeekNumber}. Weeks must be sequential.`
      );
    }

    // Business rule: Can't exceed program duration
    if (data.weekNumber > program.durationWeeks) {
      throw new ValidationError(
        `Cannot add week ${data.weekNumber}. Program duration is ${program.durationWeeks} weeks.`
      );
    }

    const week = await this.programRepo.createWeek({
      ...data,
      programId,
    });

    logger.info('Week added to program', {
      weekId: week.id,
      programId,
      weekNumber: week.weekNumber
    });

    return week;
  }

  async addDayToWeek(
    weekId: string,
    data: CreateProgramDayDto,
    userId: string
  ): Promise<ProgramDay> {
    // Get week and verify it exists
    const week = await this.programRepo.getWeek(weekId);
    if (!week) {
      throw new NotFoundError('Week not found');
    }

    // Get program and verify access (this also checks org membership)
    await this.getProgram(week.programId, userId);

    // Business rule: Auto-assign day number based on existing days
    const existingDays = await this.programRepo.getDaysForWeek(weekId);
    const nextDayNumber = existingDays.length + 1;

    const day = await this.programRepo.createDay({
      ...data,
      weekId,
      dayNumber: nextDayNumber,
    });

    logger.info('Day added to week', {
      dayId: day.id,
      weekId,
      dayNumber: day.dayNumber
    });

    return day;
  }

  async assignProgramToAthlete(
    programId: string,
    athleteId: string,
    startDate: Date,
    assignedBy: string
  ): Promise<void> {
    // Verify program exists and assigner has access
    const program = await this.getProgram(programId, assignedBy);

    // Verify assigner is a coach
    const assigner = await this.authService.getUser(assignedBy);
    this.authService.verifyCoachRole(assigner);

    // Verify athlete is in the same organization
    await this.authService.verifyOrganizationMembership(
      athleteId,
      program.organizationId
    );

    // Business rule: Athlete can't have multiple active programs
    const existingAssignment = await this.programRepo.getActiveAssignment(athleteId);
    if (existingAssignment) {
      throw new ValidationError(
        'Athlete already has an active program. Complete or remove it first.'
      );
    }

    await this.programRepo.createAssignment({
      programId,
      athleteId,
      startDate,
      assignedBy,
    });

    logger.info('Program assigned to athlete', {
      programId,
      athleteId,
      assignedBy
    });
  }
}
```

```typescript
    return res.status(403).json({ message: error.message });
```

### 3. Split Repository Layer (1,744 lines → 7 repositories)

**Priority:** P1 - Important for maintainability
**Effort:** Medium (3-4 days)
**Impact:** Medium-High - Better organization, easier to find code

#### Current Structure

```
server/
  storage.ts (1,744 lines)
    - DatabaseStorage class with 50+ methods
    - All database operations in one file
```

#### Proposed Structure

```
server/
  repositories/
    base.repository.ts           # Common repository patterns
    user.repository.ts           # User operations
    organization.repository.ts   # Organization + members
    team.repository.ts           # Team + members
    program.repository.ts        # Programs, weeks, days, exercises
    workout.repository.ts        # Workout sessions, exercise logs, sets
    exercise.repository.ts       # Exercise library
    message.repository.ts        # Messaging
```

#### Implementation Example

**`server/repositories/base.repository.ts`**:

```typescript
import { db } from '../db';
import { eq, and, desc, SQL } from 'drizzle-orm';
import { PgTable } from 'drizzle-orm/pg-core';

export abstract class BaseRepository<T extends PgTable> {
  constructor(protected table: T) {}

  async findById(id: string) {
    return await db.query[this.getTableName()].findFirst({
      where: (table: any, { eq }: any) => eq(table.id, id),
    });
  }

  async findMany(where?: SQL) {
    if (where) {
      return await db.select().from(this.table).where(where);
    }
    return await db.select().from(this.table);
  }

  async create(data: any) {
    const [result] = await db.insert(this.table).values(data).returning();
    return result;
  }

  async update(id: string, data: any) {
    const [result] = await db
      .update(this.table)
      .set({ ...data, updatedAt: new Date() })

      .returning();

  }
```

```typescript
import { BaseRepository } from './base.repository';
import {
  workoutSessions,
  exerciseLogs,
  setLogs,
  type WorkoutSession,
  type ExerciseLog,
  type SetLog,
  type InsertWorkoutSession,
  type InsertExerciseLog,
  type InsertSetLog,
} from '@shared/schema';
import { db } from '../db';
import { eq, and, desc, gte, lte } from 'drizzle-orm';

export class WorkoutRepository extends BaseRepository<typeof workoutSessions> {
  constructor() {
    super(workoutSessions);
  }

  protected getTableName(): string {
    return 'workoutSessions';
  }

  // Workout session operations
  async createSession(data: InsertWorkoutSession): Promise<WorkoutSession> {
    const [session] = await db.insert(workoutSessions).values(data).returning();
    return session;
  }

  async getSession(sessionId: string): Promise<WorkoutSession | undefined> {
    return await db.query.workoutSessions.findFirst({
      where: (sessions, { eq }) => eq(sessions.id, sessionId),
      with: {
        programDay: {
          with: {
            exercises: {
              with: {
                exercise: true,
              },
            },
          },
        },
        exerciseLogs: {
          with: {
            sets: true,
            programExercise: {
              with: {
                exercise: true,
              },
            },
          },
        },
      },
    });
  }

  async getSessionsForAthlete(
    athleteId: string,
    startDate?: Date,
    endDate?: Date
  ): Promise<WorkoutSession[]> {
    let whereClause = eq(workoutSessions.athleteId, athleteId);

    if (startDate && endDate) {
      whereClause = and(
        whereClause,
        gte(workoutSessions.date, startDate),
        lte(workoutSessions.date, endDate)
      ) as any;
    }

    return await db.query.workoutSessions.findMany({
      where: whereClause,
      orderBy: (sessions, { desc }) => [desc(sessions.date)],
    });
  }

  async getTodaySession(athleteId: string): Promise<WorkoutSession | undefined> {
    const today = new Date();
    today.setHours(0, 0, 0, 0);

    const tomorrow = new Date(today);
    tomorrow.setDate(tomorrow.getDate() + 1);

    return await db.query.workoutSessions.findFirst({
      where: (sessions, { eq, and, gte, lt }) =>
        and(
          eq(sessions.athleteId, athleteId),
          gte(sessions.date, today),
          lt(sessions.date, tomorrow)
        ),
      with: {
        programDay: {
          with: {
            exercises: {
              with: {
                exercise: true,
              },
            },
          },
        },
        exerciseLogs: {
          with: {
            sets: true,
          },
        },
      },
    });
  }

  async updateSession(
    sessionId: string,
    data: Partial<WorkoutSession>
  ): Promise<WorkoutSession> {
    const [updated] = await db
      .update(workoutSessions)
      .set({ ...data, updatedAt: new Date() })
      .where(eq(workoutSessions.id, sessionId))
      .returning();
    return updated;
  }

  // Exercise log operations
  async createExerciseLog(data: InsertExerciseLog): Promise<ExerciseLog> {
    const [log] = await db.insert(exerciseLogs).values(data).returning();
    return log;
  }

  async getExerciseLogs(sessionId: string): Promise<ExerciseLog[]> {
    return await db.query.exerciseLogs.findMany({
      where: (logs, { eq }) => eq(logs.workoutSessionId, sessionId),
      with: {
        programExercise: {
          with: {
            exercise: true,
          },
        },
        sets: {
          orderBy: (sets, { asc }) => [asc(sets.setNumber)],
        },
      },
    });
  }

  async updateExerciseLog(
    logId: string,
    data: Partial<ExerciseLog>
  ): Promise<ExerciseLog> {
    const [updated] = await db
      .update(exerciseLogs)
      .set(data)
      .where(eq(exerciseLogs.id, logId))
      .returning();
    return updated;
  }

  // Set log operations
  async createSetLog(data: InsertSetLog): Promise<SetLog> {
    const [set] = await db.insert(setLogs).values(data).returning();
    return set;
  }

  async getSetsForExercise(exerciseLogId: string): Promise<SetLog[]> {
    return await db.query.setLogs.findMany({
      where: (sets, { eq }) => eq(sets.exerciseLogId, exerciseLogId),
      orderBy: (sets, { asc }) => [asc(sets.setNumber)],
    });
  }

  async updateSetLog(setId: string, data: Partial<SetLog>): Promise<SetLog> {
    const [updated] = await db
      .update(setLogs)
      .set(data)
      .where(eq(setLogs.id, setId))
      .returning();
    return updated;
  }

  // Analytics queries
  async getAthleteStats(athleteId: string) {
    // Total workouts
    const sessions = await this.getSessionsForAthlete(athleteId);
    const completedSessions = sessions.filter(s => s.status === 'completed');

    // Calculate total volume (sum of weight * reps)
    let totalVolume = 0;
    for (const session of completedSessions) {
      const logs = await this.getExerciseLogs(session.id);
      for (const log of logs) {
        const sets = await this.getSetsForExercise(log.id);
        totalVolume += sets.reduce(
          (sum, set) => sum + (set.weight || 0) * (set.reps || 0),
          0
        );
      }
    }

    return {
      totalWorkouts: completedSessions.length,
      totalVolume,
      averageRPE:
        completedSessions.reduce((sum, s) => sum + (s.overallRpe || 0), 0) /
        completedSessions.length,
      averageDuration:
        completedSessions.reduce((sum, s) => sum + (s.duration || 0), 0) /
        completedSessions.length,
    };
  }
}
```

```typescript
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { useToast } from '@/hooks/use-toast';
import { apiRequest } from '@/lib/queryClient';

interface CreateProgramData {
  name: string;
  description?: string;
  durationWeeks: number;
  phase: string;
  organizationId: string;
}

export function useCreateProgram() {
  const queryClient = useQueryClient();
  const { toast } = useToast();

  return useMutation({
    mutationFn: (data: CreateProgramData) =>
      apiRequest('/api/programs', { method: 'POST', body: data }),

    onSuccess: (_, variables) => {
      // Invalidate programs list
      queryClient.invalidateQueries({
        queryKey: ['/api/programs', { organizationId: variables.organizationId }],
      });

      toast({
        title: 'Program created!',
        description: 'Your training program has been created successfully.',
      });
    },

    onError: (error: Error) => {
      toast({
        title: 'Failed to create program',
        description: error.message,
        variant: 'destructive',
      });
    },
  });
}
```

**Problems:**

#### Proposed Solution

**`server/services/authorization.service.ts`**:

```typescript
import { UserRepository } from '../repositories/user.repository';
import { OrganizationRepository } from '../repositories/organization.repository';
import { TeamRepository } from '../repositories/team.repository';
import { ProgramRepository } from '../repositories/program.repository';
import { NotFoundError, ForbiddenError } from '../errors';
import type { User, Organization, Team, Program } from '@shared/schema';

export class AuthorizationService {
  constructor(
    private userRepo: UserRepository,
    private orgRepo: OrganizationRepository,
    private teamRepo: TeamRepository,
    private programRepo: ProgramRepository
  ) {}

  // User operations
  async getUser(userId: string): Promise<User> {
    const user = await this.userRepo.findById(userId);
    if (!user) {
      throw new NotFoundError('User not found');
    }
    return user;
  }

  // Role checks
  verifyAdminRole(user: User): void {
    if (user.role !== 'admin') {
      throw new ForbiddenError('This action requires admin privileges');
    }
  }

  verifyCoachRole(user: User): void {
    const coachRoles = ['admin', 'head_coach', 'assistant_coach'];
    if (!coachRoles.includes(user.role)) {
      throw new ForbiddenError('This action requires coach privileges');
    }
  }

  verifyHeadCoachRole(user: User): void {
    const headCoachRoles = ['admin', 'head_coach'];
    if (!headCoachRoles.includes(user.role)) {
      throw new ForbiddenError('This action requires head coach privileges');
    }
  }

  isCoach(user: User): boolean {
    return ['admin', 'head_coach', 'assistant_coach'].includes(user.role);
  }

  isHeadCoach(user: User): boolean {
    return ['admin', 'head_coach'].includes(user.role);
  }

  // Organization authorization
  async verifyOrganizationOwnership(
    userId: string,
    orgId: string
  ): Promise<Organization> {
    const org = await this.orgRepo.findById(orgId);

    if (!org) {
      throw new NotFoundError('Organization not found');
    }

    if (org.ownerId !== userId) {
      throw new ForbiddenError(
        'You must be the organization owner to perform this action'
      );
    }

    return org;
  }

  async verifyOrganizationMembership(
    userId: string,
    orgId: string
  ): Promise<Organization> {
    const org = await this.orgRepo.findById(orgId);

    if (!org) {
      throw new NotFoundError('Organization not found');
    }

    // Owner is always a member
    if (org.ownerId === userId) {
      return org;
    }

    // Check membership
    const membership = await this.orgRepo.getMembership(orgId, userId);
    if (!membership) {
      throw new ForbiddenError(
        'You must be a member of this organization to perform this action'
      );
    }

    return org;
  }

  async verifyOrganizationCoachAccess(
    userId: string,
    orgId: string
  ): Promise<Organization> {
    const user = await this.getUser(userId);
    this.verifyCoachRole(user);

    return await this.verifyOrganizationMembership(userId, orgId);
  }

  // Team authorization
  async verifyTeamAccess(userId: string, teamId: string): Promise<Team> {
    const team = await this.teamRepo.findById(teamId);

    if (!team) {
      throw new NotFoundError('Team not found');
    }

    // Verify user has access to team's organization
    await this.verifyOrganizationMembership(userId, team.organizationId);

    // Verify user is a member of the team
    const membership = await this.teamRepo.getMembership(teamId, userId);
    if (!membership) {
      throw new ForbiddenError('You are not a member of this team');
    }

    return team;
  }

  async verifyTeamCoachAccess(userId: string, teamId: string): Promise<Team> {
    const team = await this.verifyTeamAccess(userId, teamId);

    // Additional check: is user a coach on this team?
    const membership = await this.teamRepo.getMembership(teamId, userId);
    if (membership && !this.isCoach({ role: membership.role } as User)) {
      throw new ForbiddenError('This action requires coach privileges on this team');
    }

    return team;
  }

  // Program authorization
  async verifyProgramAccess(userId: string, programId: string): Promise<Program> {
    const program = await this.programRepo.findById(programId);

    if (!program) {
      throw new NotFoundError('Program not found');
    }

    // Verify user has access to program's organization
    await this.verifyOrganizationMembership(userId, program.organizationId);

    return program;
  }

  async verifyProgramModifyAccess(
    userId: string,
    programId: string
  ): Promise<Program> {
    const program = await this.verifyProgramAccess(userId, programId);
    const user = await this.getUser(userId);

    // Only coaches or program creator can modify
    if (program.createdBy !== userId && !this.isCoach(user)) {
      throw new ForbiddenError(
        'Only coaches or the program creator can modify programs'
      );
    }

    return program;
  }

  // Workout authorization
  async verifyWorkoutSessionAccess(
    userId: string,
    sessionId: string
  ): Promise<any> {
    // Implementation depends on workout session structure
    // Similar pattern to above
  }

  // Helper: Check if user can manage another user
  canManageUser(actor: User, target: User): boolean {
    // Admins can manage anyone
    if (actor.role === 'admin') return true;

    // Head coaches can manage assistant coaches and athletes
    if (actor.role === 'head_coach') {
      return ['assistant_coach', 'athlete'].includes(target.role);
    }

    // Assistant coaches can't manage anyone
    return false;
  }

  // Helper: Get effective permissions for a user in an organization
  async getUserPermissions(userId: string, orgId: string) {
    const org = await this.verifyOrganizationMembership(userId, orgId);
    const user = await this.getUser(userId);

    const isOwner = org.ownerId === userId;
    const isAdmin = user.role === 'admin';
    const isHeadCoach = user.role === 'head_coach';
    const isCoach = this.isCoach(user);

    return {
      canCreatePrograms: isCoach,
      canDeletePrograms: isOwner || isAdmin,
      canManageTeams: isCoach,
      canInviteMembers: isCoach,
      canManageOrganization: isOwner || isAdmin,
      canAssignPrograms: isCoach,
      canViewAllWorkouts: isCoach,
      canModifySettings: isOwner,
    };
  }
}
```

```typescript
    }

    await this.programRepo.delete(programId);

    programId: string,
    data: UpdateProgramDto,
    userId: string
  ) {
    // Single line for authorization!

    // Business logic
    return await this.programRepo.update(programId, data);
  }
}
```

```typescript
export function useLogSet() {
  const queryClient = useQueryClient();

  return useMutation({
    mutationFn: (data: LogSetData) =>
      apiRequest(`/api/exercise-logs/${data.exerciseLogId}/sets`, {
        method: 'POST',
        body: data,
      }),

    // Optimistic update - immediately show in UI

      await queryClient.cancelQueries({
        queryKey: ['/api/workout-sessions/today']
      });

      // Snapshot current value
      const previous = queryClient.getQueryData(['/api/workout-sessions/today']);

      // Optimistically update
      queryClient.setQueryData(['/api/workout-sessions/today'], (old: any) => {
        // Add new set to the session
        return {
          ...old,
          // ... update logic
        };
      });

      // Return snapshot for rollback
      return { previous };
    },

    // Rollback on error
    onError: (err, newSet, context) => {
      queryClient.setQueryData(
        ['/api/workout-sessions/today'],
        context?.previous
      );

      toast({
        title: 'Failed to log set',
        description: err.message,
        variant: 'destructive',
      });
    },

    // Refetch to ensure consistency
    onSettled: () => {
      queryClient.invalidateQueries({
        queryKey: ['/api/workout-sessions/today']
      });
    },
  });
```

```typescript
export class Program {
  constructor(
    public readonly id: string,
    public readonly name: string,
    public readonly description: string | null,
    public readonly durationWeeks: number,
    public readonly phase: ProgramPhase,
    public readonly organizationId: string,
    public readonly createdBy: string,
    public readonly createdAt: Date,
    public readonly updatedAt: Date,
    private weeks: Week[] = []
  ) {}

  // Business logic methods
  addWeek(week: Week): void {
    if (this.weeks.length >= this.durationWeeks) {
      throw new Error('Program is at maximum duration');
    }
    this.weeks.push(week);
  }

  canBeEditedBy(userId: string, userRole: UserRole): boolean {
    return this.createdBy === userId ||
           ['admin', 'head_coach'].includes(userRole);
  }

  canBeDeletedBy(userId: string, orgOwnerId: string): boolean {
    return this.createdBy === userId || userId === orgOwnerId;
  }

  isComplete(): boolean {
    return this.weeks.length === this.durationWeeks &&
           this.weeks.every(week => week.isComplete());
  }

  getTotalDays(): number {
    return this.weeks.reduce((sum, week) => sum + week.days.length, 0);
  }

  getTotalExercises(): number {
    return this.weeks.reduce(
      (sum, week) => sum + week.getTotalExercises(),
      0
    );
  }

  getProgress(): number {
    if (this.weeks.length === 0) return 0;
    const completedWeeks = this.weeks.filter(w => w.isComplete()).length;
    return (completedWeeks / this.durationWeeks) * 100;
  }

  // Serialization for API responses
  toJSON() {
    return {
      id: this.id,
      name: this.name,
      description: this.description,
      durationWeeks: this.durationWeeks,
      phase: this.phase,
      organizationId: this.organizationId,
      createdBy: this.createdBy,
      createdAt: this.createdAt,
      updatedAt: this.updatedAt,
      // Computed properties
      totalDays: this.getTotalDays(),
      totalExercises: this.getTotalExercises(),
      progress: this.getProgress(),
      isComplete: this.isComplete(),
      // Include nested data if loaded
      weeks: this.weeks.map(w => w.toJSON()),
    };
  }

  // Factory method from database entity
  static fromEntity(entity: ProgramEntity): Program {
    return new Program(
      entity.id,
      entity.name,
      entity.description,
      entity.durationWeeks,
      entity.phase,
      entity.organizationId,
      entity.createdBy,
      entity.createdAt,
      entity.updatedAt
    );
  }
}
```

```typescript
import { persist, createJSONStorage } from 'zustand/middleware';

interface AppState {
  // Only store IDs, not full objects
  selectedOrganizationId: string | null;
  selectedTeamId: string | null;

  // UI state
  sidebarCollapsed: boolean;
  theme: 'light' | 'dark' | 'system';

  // Actions
  setSelectedOrganization: (id: string | null) => void;
  setSelectedTeam: (id: string | null) => void;
  setSidebarCollapsed: (collapsed: boolean) => void;
  setTheme: (theme: 'light' | 'dark' | 'system') => void;

  // Reset
  reset: () => void;
}

  selectedOrganizationId: null,
  selectedTeamId: null,
  sidebarCollapsed: false,
  theme: 'system' as const,
};

export const useAppStore = create<AppState>()(
  persist(
    (set) => ({
      ...initialState,

      setSelectedOrganization: (id) => set({ selectedOrganizationId: id }),
      setSelectedTeam: (id) => set({ selectedTeamId: id }),
      setSidebarCollapsed: (collapsed) => set({ sidebarCollapsed: collapsed }),
      setTheme: (theme) => set({ theme }),

      reset: () => set(initialState),
    }),
    {
      name: 'app-state',
      storage: createJSONStorage(() => localStorage),
    }
  )
);
```

### 9. Event-Driven Architecture

**Priority:** P3 - Future enhancement
**Effort:** High (1-2 weeks)
**Impact:** Medium - Better extensibility

#### Current Approach

Everything is synchronous and direct:


**Problems:**
- Tightly coupled side effects
- Hard to add new features without changing core logic
- If notification fails, entire operation fails

#### Event-Driven Approach

**`server/events/event-bus.ts`**:

```typescript
import { EventEmitter } from 'events';

export interface DomainEvent {
  eventType: string;
  occurredAt: Date;
  aggregateId: string;
  data: any;
}

class EventBus extends EventEmitter {
  private static instance: EventBus;

  private constructor() {
    super();
    this.setMaxListeners(100); // Allow many listeners
  }

  static getInstance(): EventBus {
    if (!EventBus.instance) {
      EventBus.instance = new EventBus();
    }
    return EventBus.instance;
  }

  publish(event: DomainEvent): void {
    this.emit(event.eventType, event);
    logger.debug('Event published', { eventType: event.eventType });
  }

  subscribe(eventType: string, handler: (event: DomainEvent) => Promise<void>): void {
    this.on(eventType, async (event) => {
      try {
        await handler(event);
      } catch (error) {
        logger.error(`Error handling event ${eventType}`, error);
      }
    });
  }
}

export const eventBus = EventBus.getInstance();
```

**Priority:** P4 - Only if analytics become complex
**Effort:** High (2 weeks)
**Impact:** Low-Medium - Better query performance

#### When to Use

If you find yourself writing complex aggregate queries like:

```sql
SELECT
  u.id,
  u.name,
  COUNT(DISTINCT ws.id) as total_workouts,
  SUM(sl.weight * sl.reps) as total_volume,
  AVG(ws.overall_rpe) as avg_rpe,
  COUNT(DISTINCT pa.id) as programs_completed
FROM users u
LEFT JOIN workout_sessions ws ON ws.athlete_id = u.id
LEFT JOIN exercise_logs el ON el.workout_session_id = ws.id
LEFT JOIN set_logs sl ON sl.exercise_log_id = el.id

GROUP BY u.id;
```

#### CQRS Approach

**Write Model** (Commands):
```typescript
workoutService.completeWorkout(sessionId); // Writes to workout_sessions
```

**Read Model** (Queries):
```typescript
// Separate optimized table for analytics
athlete_statistics:
  - athlete_id
  - total_workouts
  - total_volume
  - avg_rpe
  - last_workout_date
  - updated_at
```

**Event Handler Updates Read Model:**
```typescript
eventBus.subscribe('workout.completed', async (event) => {
  // Update denormalized statistics table
  await db.execute(`
    UPDATE athlete_statistics
    SET
      total_workouts = total_workouts + 1,
      total_volume = total_volume + ${event.data.totalVolume},

```

### 12. Background Jobs (BullMQ)

**Priority:** P4 - When you have long-running operations
**Effort:** High (1 week)
**Impact:** Medium - Better UX for bulk operations

#### When to Use

- Bulk program assignments (assigning to 100 athletes)
- Email sending (bulk emails to team)
- Report generation (export workout history to PDF)
- Data migrations

#### Implementation

**Install:**
```bash
npm install bullmq ioredis
```

**`server/jobs/queue.ts`**:

```typescript
import { Queue, Worker } from 'bullmq';
import Redis from 'ioredis';

const connection = new Redis({
  host: process.env.REDIS_HOST || 'localhost',
  port: parseInt(process.env.REDIS_PORT || '6379'),
  maxRetriesPerRequest: null,
});

export const programQueue = new Queue('program-operations', { connection });

// Worker
const worker = new Worker(
  'program-operations',
  async (job) => {
    if (job.name === 'assign-program-bulk') {
      const { programId, athleteIds, assignedBy } = job.data;

      for (let i = 0; i < athleteIds.length; i++) {
        await programService.assignProgram(
          programId,
          athleteIds[i],
          new Date(),
          assignedBy
        );

        // Update progress
        await job.updateProgress((i + 1) / athleteIds.length * 100);
      }

      return { assigned: athleteIds.length };
    }
  },
  { connection }
);

worker.on('completed', (job) => {
  logger.info('Job completed', { jobId: job.id });
});

worker.on('failed', (job, err) => {
  logger.error('Job failed', { jobId: job?.id, error: err });
});
```

**Usage:**

```typescript
// routes/programs.routes.ts
app.post('/api/program-assignments/bulk', isAuthenticated, async (req, res) => {
  const { programId, athleteIds } = req.body;

  // Queue job instead of blocking
  const job = await programQueue.add('assign-program-bulk', {
    programId,
    athleteIds,
    assignedBy: req.currentUser!.id,
  });

  res.json({
    message: 'Assignment job queued',
    jobId: job.id,
  });
});

// Check job status
app.get('/api/jobs/:jobId', isAuthenticated, async (req, res) => {
  const job = await programQueue.getJob(req.params.jobId);

  if (!job) {
    return res.status(404).json({ message: 'Job not found' });
  }

  const state = await job.getState();
  const progress = job.progress;

  res.json({
    jobId: job.id,
    state,
    progress,
  });
});
```

**Benefits:**
- ✅ Don't block HTTP requests
- ✅ Retry failed operations
- ✅ Track progress
- ✅ Scalable (multiple workers)
