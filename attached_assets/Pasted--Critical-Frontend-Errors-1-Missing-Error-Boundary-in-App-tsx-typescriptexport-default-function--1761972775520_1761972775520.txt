2. Race Condition in AppContext.tsx
typescriptuseEffect(() => {
  const storedOrg = localStorage.getItem('currentOrganization');
  const storedTeam = localStorage.getItem('currentTeam');

  if (storedOrg) setCurrentOrganization(JSON.parse(storedOrg));
  if (storedTeam) setCurrentTeam(JSON.parse(storedTeam));
  
  localStorage.removeItem('currentUser'); // ❌ Happens async
}, []);
Problem: Multiple setCurrentOrganization calls trigger multiple re-renders. Also, JSON.parse can throw if data is corrupted.
Fix:
typescriptuseEffect(() => {
  try {
    const storedOrg = localStorage.getItem('currentOrganization');
    const storedTeam = localStorage.getItem('currentTeam');

    const org = storedOrg ? JSON.parse(storedOrg) : null;
    const team = storedTeam ? JSON.parse(storedTeam) : null;
    
    // Batch state updates
    if (org) setCurrentOrganization(org);
    if (team) setCurrentTeam(team);
  } catch (error) {
    console.error('Failed to parse stored data:', error);
    localStorage.removeItem('currentOrganization');
    localStorage.removeItem('currentTeam');
  } finally {
    localStorage.removeItem('currentUser');
  }
}, []);

typescriptasync function throwIfResNotOk(res: Response) {
  if (!res.ok) {
    let text = res.statusText;
    try {
      text = await res.text();
    } catch (error) {
      console.error('Failed to read error response:', error);
    }
    throw new Error(`${res.status}: ${text || 'Request failed'}`);
