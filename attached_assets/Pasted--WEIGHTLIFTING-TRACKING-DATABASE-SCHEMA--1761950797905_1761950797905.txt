-- Core Exercise Library Tables
-- ============================================

CREATE TABLE exercise_categories (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Insert main categories
INSERT INTO exercise_categories (name, description) VALUES
('Clean', 'Clean variations and related movements'),
('Snatch', 'Snatch variations and related movements'),
('Jerk', 'Jerk variations and related movements'),
('Squat', 'Squat variations'),
('Plyo', 'Plyometric exercises'),
('Accessory', 'Accessory movements'),
('Isometric', 'Isometric holds'),
('Competition', 'Competition lifts'),
('Pull', 'Pulling movements'),
('Variation', 'Movement variations');

CREATE TABLE exercise_secondary_tags (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50) NOT NULL UNIQUE,
    description TEXT
);

-- Insert secondary tags
INSERT INTO exercise_secondary_tags (name, description) VALUES
('Lower', 'Lower body focus'),
('Upper', 'Upper body focus'),
('Pull', 'Pulling movement'),
('Variation', 'Variation of main lift'),
('Accessory', 'Accessory movement'),
('Competition', 'Competition movement');

CREATE TABLE exercises (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE,
    primary_category_id INTEGER REFERENCES exercise_categories(id),
    secondary_tag_id INTEGER REFERENCES exercise_secondary_tags(id),
    youtube_url TEXT,
    description TEXT,
    difficulty_level VARCHAR(20) CHECK (difficulty_level IN ('Beginner', 'Intermediate', 'Advanced', 'Elite')),
    is_competition_lift BOOLEAN DEFAULT FALSE,
    requires_equipment TEXT[], -- Array of required equipment
    common_faults TEXT[], -- Common technical faults to watch for
    coaching_cues TEXT[], -- Key coaching points
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Create index for faster searches
CREATE INDEX idx_exercises_primary_category ON exercises(primary_category_id);
CREATE INDEX idx_exercises_secondary_tag ON exercises(secondary_tag_id);
CREATE INDEX idx_exercises_name ON exercises(name);

-- Athlete Management Tables
-- ============================================

CREATE TABLE athletes (
    id SERIAL PRIMARY KEY,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE,
    date_of_birth DATE,
    gender VARCHAR(20),
    weight_class VARCHAR(20), -- e.g., '73kg', '81kg', etc.
    current_bodyweight DECIMAL(5,2),
    competition_bodyweight DECIMAL(5,2),
    experience_level VARCHAR(20) CHECK (experience_level IN ('Novice', 'Intermediate', 'Advanced', 'Elite')),
    coach_notes TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_athletes_active ON athletes(is_active);
CREATE INDEX idx_athletes_weight_class ON athletes(weight_class);

CREATE TABLE athlete_personal_records (
    id SERIAL PRIMARY KEY,
    athlete_id INTEGER REFERENCES athletes(id) ON DELETE CASCADE,
    exercise_id INTEGER REFERENCES exercises(id),
    weight_kg DECIMAL(6,2) NOT NULL,
    reps INTEGER DEFAULT 1,
    date_achieved DATE NOT NULL,
    video_url TEXT,
    notes TEXT,
    verified_by VARCHAR(100), -- Coach who verified
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(athlete_id, exercise_id, date_achieved)
);

CREATE INDEX idx_pr_athlete ON athlete_personal_records(athlete_id);
CREATE INDEX idx_pr_exercise ON athlete_personal_records(exercise_id);
CREATE INDEX idx_pr_date ON athlete_personal_records(date_achieved);

-- Training Program Tables
-- ============================================

CREATE TABLE training_cycles (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    focus TEXT, -- e.g., "Strength Phase", "Competition Prep", "Off-Season"
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE training_sessions (
    id SERIAL PRIMARY KEY,
    athlete_id INTEGER REFERENCES athletes(id) ON DELETE CASCADE,
    cycle_id INTEGER REFERENCES training_cycles(id),
    session_date DATE NOT NULL,
    session_type VARCHAR(50), -- e.g., "Heavy Day", "Technique", "Accessory"
    planned_duration_minutes INTEGER,
    actual_duration_minutes INTEGER,
    rpe_rating INTEGER CHECK (rpe_rating BETWEEN 1 AND 10), -- Rate of Perceived Exertion
    readiness_score INTEGER CHECK (readiness_score BETWEEN 1 AND 10),
    bodyweight DECIMAL(5,2),
    notes TEXT,
    completed BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_sessions_athlete ON training_sessions(athlete_id);
CREATE INDEX idx_sessions_date ON training_sessions(session_date);
CREATE INDEX idx_sessions_cycle ON training_sessions(cycle_id);

CREATE TABLE session_exercises (
    id SERIAL PRIMARY KEY,
    session_id INTEGER REFERENCES training_sessions(id) ON DELETE CASCADE,
    exercise_id INTEGER REFERENCES exercises(id),
    exercise_order INTEGER, -- Order in the workout
    sets_planned INTEGER,
    sets_completed INTEGER,
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_session_exercises_session ON session_exercises(session_id);

CREATE TABLE exercise_sets (
    id SERIAL PRIMARY KEY,
    session_exercise_id INTEGER REFERENCES session_exercises(id) ON DELETE CASCADE,
    set_number INTEGER NOT NULL,
    weight_kg DECIMAL(6,2),
    reps_planned INTEGER,
    reps_completed INTEGER,
    rpe INTEGER CHECK (rpe BETWEEN 1 AND 10),
    bar_speed DECIMAL(4,2), -- meters/second if using velocity tracking
    is_pr BOOLEAN DEFAULT FALSE,
    form_rating INTEGER CHECK (form_rating BETWEEN 1 AND 5), -- 5 = perfect
    video_url TEXT,
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_sets_session_exercise ON exercise_sets(session_exercise_id);

-- Competition Management
-- ============================================

    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    date DATE NOT NULL,
    location VARCHAR(255),
    competition_level VARCHAR(50), -- e.g., "Local", "State", "National", "International"
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE competition_results (
    id SERIAL PRIMARY KEY,
    athlete_id INTEGER REFERENCES athletes(id) ON DELETE CASCADE,
    competition_id INTEGER REFERENCES competitions(id) ON DELETE CASCADE,
    weight_class VARCHAR(20),
    bodyweight DECIMAL(5,2),
    
    -- Snatch attempts
    snatch_1 DECIMAL(6,2),
    snatch_1_result VARCHAR(10) CHECK (snatch_1_result IN ('good', 'no lift', 'withdrawn')),
    snatch_2 DECIMAL(6,2),
    snatch_2_result VARCHAR(10) CHECK (snatch_2_result IN ('good', 'no lift', 'withdrawn')),
    snatch_3 DECIMAL(6,2),
    snatch_3_result VARCHAR(10) CHECK (snatch_3_result IN ('good', 'no lift', 'withdrawn')),
    best_snatch DECIMAL(6,2),
    
    -- Clean & Jerk attempts
    cj_1 DECIMAL(6,2),
    cj_1_result VARCHAR(10) CHECK (cj_1_result IN ('good', 'no lift', 'withdrawn')),
    cj_2 DECIMAL(6,2),
    cj_2_result VARCHAR(10) CHECK (cj_2_result IN ('good', 'no lift', 'withdrawn')),
    cj_3 DECIMAL(6,2),
    cj_3_result VARCHAR(10) CHECK (cj_3_result IN ('good', 'no lift', 'withdrawn')),
    best_cj DECIMAL(6,2),
    
    total DECIMAL(6,2),
    sinclair DECIMAL(6,3), -- Sinclair coefficient
    placement INTEGER,
    
    notes TEXT,
    video_url TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_comp_results_athlete ON competition_results(athlete_id);
CREATE INDEX idx_comp_results_competition ON competition_results(competition_id);

-- Useful Views for Analytics
-- ============================================

-- View: Athlete Progress Overview
CREATE VIEW athlete_progress_summary AS
SELECT 
    a.id as athlete_id,
    a.first_name || ' ' || a.last_name as athlete_name,
    COUNT(DISTINCT ts.id) as total_sessions,
    COUNT(DISTINCT DATE_TRUNC('week', ts.session_date)) as training_weeks,
    AVG(ts.rpe_rating) as avg_rpe,
    COUNT(apr.id) as total_prs,
    MAX(ts.session_date) as last_training_date
FROM athletes a
LEFT JOIN training_sessions ts ON a.id = ts.athlete_id
LEFT JOIN athlete_personal_records apr ON a.id = apr.athlete_id
WHERE a.is_active = TRUE
GROUP BY a.id, a.first_name, a.last_name;

-- View: Exercise Volume Trends
CREATE VIEW exercise_volume_trends AS
SELECT 
    a.id as athlete_id,
    a.first_name || ' ' || a.last_name as athlete_name,
    e.name as exercise_name,
    DATE_TRUNC('week', ts.session_date) as week,
    COUNT(es.set_number) as total_sets,
    SUM(es.weight_kg * es.reps_completed) as total_volume,
    AVG(es.weight_kg) as avg_weight,
    AVG(es.rpe) as avg_rpe
FROM athletes a
JOIN training_sessions ts ON a.id = ts.athlete_id
JOIN session_exercises se ON ts.id = se.session_id
JOIN exercises e ON se.exercise_id = e.id
JOIN exercise_sets es ON se.id = es.session_exercise_id
WHERE es.reps_completed IS NOT NULL
GROUP BY a.id, a.first_name, a.last_name, e.name, DATE_TRUNC('week', ts.session_date);

-- View: Competition Performance Tracking
CREATE VIEW competition_performance AS
SELECT 
    a.id as athlete_id,
    a.first_name || ' ' || a.last_name as athlete_name,
    c.name as competition_name,
    c.date as competition_date,
    cr.weight_class,
    cr.best_snatch,
    cr.best_cj,
    cr.total,
    cr.sinclair,
    cr.placement,
    LAG(cr.total) OVER (PARTITION BY a.id ORDER BY c.date) as previous_total,
    cr.total - LAG(cr.total) OVER (PARTITION BY a.id ORDER BY c.date) as total_improvement
FROM athletes a
JOIN competition_results cr ON a.id = cr.athlete_id
JOIN competitions c ON cr.competition_id = c.id
ORDER BY a.id, c.date;

-- Functions for Common Queries
-- ============================================

-- Function: Get athlete's best lifts for each exercise
CREATE OR REPLACE FUNCTION get_athlete_best_lifts(p_athlete_id INTEGER)
RETURNS TABLE (
    exercise_name VARCHAR,
    best_weight DECIMAL,
    date_achieved DATE,
    reps INTEGER
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        e.name,
        apr.weight_kg,
        apr.date_achieved,
        apr.reps
    FROM athlete_personal_records apr
    JOIN exercises e ON apr.exercise_id = e.id
    WHERE apr.athlete_id = p_athlete_id
    ORDER BY e.name;
END;
$$ LANGUAGE plpgsql;

-- Function: Calculate weekly training volume
CREATE OR REPLACE FUNCTION calculate_weekly_volume(
    p_athlete_id INTEGER,
    p_start_date DATE,
    p_end_date DATE
)
RETURNS TABLE (
    week_start DATE,
    total_volume DECIMAL,
    total_reps INTEGER,
    session_count INTEGER
) AS $$
BEGIN
    RETURN QUERY
    SELECT 
        DATE_TRUNC('week', ts.session_date)::DATE,
        SUM(es.weight_kg * es.reps_completed),
        SUM(es.reps_completed)::INTEGER,
        COUNT(DISTINCT ts.id)::INTEGER
    FROM training_sessions ts
    JOIN session_exercises se ON ts.id = se.session_id
    JOIN exercise_sets es ON se.id = es.session_exercise_id
    WHERE ts.athlete_id = p_athlete_id
        AND ts.session_date BETWEEN p_start_date AND p_end_date
        AND es.reps_completed IS NOT NULL
    GROUP BY DATE_TRUNC('week', ts.session_date)
    ORDER BY DATE_TRUNC('week', ts.session_date);
END;
$$ LANGUAGE plpgsql;

-- Triggers for automatic timestamp updates
-- ============================================

CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_athletes_updated_at BEFORE UPDATE ON athletes
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_exercises_updated_at BEFORE UPDATE ON exercises
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_training_sessions_updated_at BEFORE UPDATE ON training_sessions
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();
