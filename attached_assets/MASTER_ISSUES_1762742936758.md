### 7. N+1 Query Problem - Teams with Members
**Severity:** HIGH - Performance Issue
**File:** `server/routes.ts`
**Lines:** 276-283

**Issue:**
```typescript
const teamsWithMembers = await Promise.all(
  teams.map(async (team) => {
    const allMembers = await storage.getTeamMembers(team.id);  // N+1 QUERY
    const athleteMembers = allMembers.filter(m => m.user.role === 'athlete');
    return { ...team, members: athleteMembers };
  })
);
```

**Problem:**
- Executes 1 query to fetch teams + N queries for members (one per team)
- If you have 10 teams, this executes 11 database queries
- Scales poorly with number of teams

**Impact:**
- Slow response times (10 teams = ~200ms overhead)
- Database connection pool exhaustion
- Poor scalability

**Fix:**
Create new method in `server/storage.ts`:
```typescript
async getTeamsWithMembers(organizationId: string): Promise<Array<Team & { members: User[] }>> {
  const results = await db
    .select()
    .from(teams)
    .leftJoin(teamMembers, eq(teams.id, teamMembers.teamId))
    .leftJoin(users, eq(teamMembers.userId, users.id))
    .where(eq(teams.organizationId, organizationId));

  // Group results by team
  const teamMap = new Map();
  for (const row of results) {
    if (!teamMap.has(row.teams.id)) {
      teamMap.set(row.teams.id, { ...row.teams, members: [] });
    }
    if (row.users && row.users.role === 'athlete') {
      teamMap.get(row.teams.id).members.push(row.users);
    }
  }
  return Array.from(teamMap.values());
}
```

### 10. Race Condition in Join Request Duplicate Check
**Severity:** HIGH - Data Integrity Issue
**File:** `server/routes.ts`
**Lines:** 370-375, 602-620

**Issue:**
```typescript
// Check if user already has a pending request (TOCTOU)
const existingRequests = await storage.getUserJoinRequests(req.currentUser!.id, 'pending');
const hasPendingRequest = existingRequests.some(r => r.teamId === req.body.teamId);
if (hasPendingRequest) {
  return res.status(400).json({ message: "Already has pending request" });
```

**Problem:**
- Time-of-check to time-of-use (TOCTOU) race condition
- Two concurrent requests can both pass the check
- Both requests get created, violating uniqueness

**Impact:**
- Duplicate join requests possible
- Database constraint violations in production
- Poor user experience

**Fix:**
```typescript
// Remove the duplicate check, rely on database constraint
try {
  const request = await storage.createTeamJoinRequest(data);
  res.json(request);
} catch (error) {
  if (error.code === '23505') { // PostgreSQL unique violation
    return res.status(400).json({
      message: "You already have a pending request for this team"
    });
  }
  throw error;
}
```

**Impact:**

**Fix:**

```typescript
import rateLimit from 'express-rate-limit';
import RedisStore from 'rate-limit-redis';
import Redis from 'ioredis';

const redis = new Redis(process.env.REDIS_URL);

export const createRateLimiter = (options: {
  windowMs: number;
  max: number;
  message?: string;
}) => rateLimit({
  store: new RedisStore({
    client: redis,
    prefix: 'rl:',
  }),
  windowMs: options.windowMs,
  max: options.max,
  message: options.message || 'Too many requests',
  standardHeaders: true,
  legacyHeaders: false,
  // Use both IP and user ID if authenticated
  keyGenerator: (req) => {
    const user = (req as any).currentUser;
    return user ? `user:${user.id}` : `ip:${req.ip}`;
  },
});

// Apply to multiple endpoints
export const authRateLimit = createRateLimiter({
  windowMs: 15 * 60 * 1000,
  max: 5,
  message: 'Too many authentication attempts'
});

export const apiRateLimit = createRateLimiter({
  windowMs: 60 * 1000,
  max: 100,
  message: 'Too many API requests'
});
```

```typescript
  if (!req.isAuthenticated() || !req.user) {
    return res.status(401).json({ message: "Unauthorized" });
  }

  // Check if user context already in session
  if (req.session.userContext) {
    (req as any).currentUser = req.session.userContext;

  }

  const userId = (req.user as User).id;
```

**Option 2: Redis Cache**
```typescript
export const isAuthenticated: RequestHandler = async (req, res, next) => {
  if (!req.isAuthenticated() || !req.user) {
    return res.status(401).json({ message: "Unauthorized" });
  }

  const userId = (req.user as User).id;
  const cacheKey = `user:context:${userId}`;
  
  // Try cache first
  let userContext = await redis.get(cacheKey);
  if (userContext) {
    (req as any).currentUser = JSON.parse(userContext);
    return next();
  }
  
  // Cache miss - fetch and cache for 5 minutes
  const fullUser = await storage.getUser(userId);
  const teams = await storage.getUserTeams(userId);
  
  userContext = {
    ...fullUser,
    teams: teams,
    organizationIds: [...new Set(teams.map(t => t.organizationId))],
  };
  
  await redis.setex(cacheKey, 300, JSON.stringify(userContext)); // 5 min TTL

  return next();
};
```

```typescript
export class VBTImporter {
  /**
   * Auto-detect CSV format from headers
   */
  static detectFormat(headers: string[]): 'vitruve' | 'custom' | 'gymaware' | 'unknown' {
    const headerSet = new Set(headers.map(h => h.toLowerCase().trim()));
    
    // Check for Vitruve signature columns
    if (
      headerSet.has('mean propulsive velocity (m/s)') &&
      headerSet.has('peak velocity (m/s)') &&
      headerSet.has('fatigue (pv) (%)')
    ) {
      return 'vitruve';
    }
    
    // Check for GymAware signature
    if (headerSet.has('set peak velocity') && headerSet.has('average velocity')) {
      return 'gymaware';
    }
    
    // Custom or unknown
    return headerSet.size > 5 ? 'custom' : 'unknown';
  }
  
  /**
   * Parse Vitruve CSV file
   */
  static async parseVitruve(fileBuffer: Buffer): Promise<VitruveRow[]> {
    return new Promise((resolve, reject) => {
      Papa.parse(fileBuffer.toString(), {
        header: true,
        skipEmptyLines: true,
        transformHeader: (header) => header.trim(),
        complete: (results) => {
          try {
            const parsed = results.data.map((row, idx) => {
              try {
                return vitruveRowSchema.parse(row);
              } catch (err) {
                throw new Error(`Row ${idx + 2} validation failed: ${err.message}`);
              }
            });
            resolve(parsed);
          } catch (error) {
            reject(error);
          }
        },
        error: (error) => reject(error),
      });
    });
  }
  
  /**
   * Group reps into sets
   */
  static groupBySet(rows: VitruveRow[]): Map<number, VitruveRow[]> {
    const setMap = new Map<number, VitruveRow[]>();
    
    for (const row of rows) {
      const setNum = row['# Set'];
      if (!setMap.has(setNum)) {
        setMap.set(setNum, []);
      }
      setMap.get(setNum)!.push(row);
    }
    
    return setMap;
  }
  
  /**
   * Calculate set-level averages from rep data
   */
  static calculateSetAverages(reps: VitruveRow[]) {
    const validReps = reps.filter(r => r['Type*'] === 'concentric');
    const count = validReps.length;
    
    if (count === 0) return null;
    
    const sum = (field: keyof VitruveRow) =>
      validReps.reduce((acc, r) => acc + (Number(r[field]) || 0), 0);
    
    return {
      meanPropulsiveVelocity: sum('Mean Propulsive Velocity (m/s)') / count,
      meanVelocity: sum('Mean Velocity (m/s)') / count,
      peakVelocity: Math.max(...validReps.map(r => Number(r['Peak Velocity (m/s)']) || 0)),
      meanForce: sum('Mean Force (N)') / count,
      peakForce: Math.max(...validReps.map(r => Number(r['Peak Force (N)']) || 0)),
      meanPowerMPV: sum('Mean Power [MPV] (W)') / count,
      meanPowerMV: sum('Mean Power [MV] (W)') / count,
      peakPower: Math.max(...validReps.map(r => Number(r['Peak Power (W)']) || 0)),
      velocityLossMPV: validReps[validReps.length - 1]?.[' Fatigue [MPV] (%)'],
      rom: sum('ROM (Range of Motion) (m)') / count,
      weight: validReps[0]['Weight (kg)'],
      reps: count,
    };
  }
  
  /**
   * Import Vitruve data to database
   */
  static async importVitruve(
    userId: string,
    workoutSessionId: string,
    fileBuffer: Buffer,
    fileName: string
  ): Promise<{
    success: boolean;
    setsImported: number;
    repsImported: number;
    errors: string[];
  }> {
    const errors: string[] = [];
    let setsImported = 0;
    let repsImported = 0;
    
    try {
      // 1. Parse CSV
      const rows = await this.parseVitruve(fileBuffer);
      
      if (rows.length === 0) {
        throw new Error('No data rows found in CSV');
      }
      
      // 2. Group by exercise and set
      const exerciseGroups = new Map<string, VitruveRow[]>();
      for (const row of rows) {
        const exercise = row.Exercise;
        if (!exerciseGroups.has(exercise)) {
          exerciseGroups.set(exercise, []);
        }
        exerciseGroups.get(exercise)!.push(row);
      }
      
      // 3. Import in transaction
      await db.transaction(async (tx) => {
        for (const [exerciseName, exerciseRows] of exerciseGroups) {
          // Find or create exercise
          let exercise = await tx
            .select()
            .from(exercises)
            .where(eq(exercises.name, exerciseName))
            .limit(1);
          
          if (exercise.length === 0) {
            // Create exercise if doesn't exist
            [exercise[0]] = await tx
              .insert(exercises)
              .values({
                name: exerciseName,
                category: 'imported',
                createdBy: userId,
              })
              .returning();
          }
          
          // Create exercise log for session
          const [exerciseLog] = await tx
            .insert(exerciseLogs)
            .values({
              sessionId: workoutSessionId,
              exerciseId: exercise[0].id,
              order: 0, // TODO: determine order
            })
            .returning();
          
          // Group reps by set
          const setGroups = this.groupBySet(exerciseRows);
          
          for (const [setNum, setReps] of setGroups) {
            // Calculate set averages
            const setData = this.calculateSetAverages(setReps);
            
            if (!setData) {
              errors.push(`Set ${setNum} for ${exerciseName} has no valid reps`);
              continue;
            }
            
            // Create set log
            const [setLog] = await tx
              .insert(setLogs)
              .values({
                exerciseLogId: exerciseLog.id,
                setNumber: setNum,
                weight: setData.weight,
                reps: setData.reps,
                
                // VBT data
                meanPropulsiveVelocity: setData.meanPropulsiveVelocity,
                meanVelocity: setData.meanVelocity,
                peakVelocity: setData.peakVelocity,
                meanForce: setData.meanForce,
                peakForce: setData.peakForce,
                meanPowerMPV: setData.meanPowerMPV,
                meanPowerMV: setData.meanPowerMV,
                peakPower: setData.peakPower,
                velocityLossMPV: setData.velocityLossMPV,
                rom: setData.rom,
                
                // Import metadata
                importedFrom: 'vitruve',
                importedAt: new Date(),
              })
              .returning();
            
            setsImported++;
            
            // Create rep logs
            for (const rep of setReps) {
              if (rep['Type*'] !== 'concentric') continue; // Skip eccentric for now
              
              await tx.insert(repLogs).values({
                setLogId: setLog.id,
                repNumber: rep['# Rep.'],
                movementType: rep['Type*'],
                
                meanPropulsiveVelocity: rep['Mean Propulsive Velocity (m/s)'],
                meanVelocity: rep['Mean Velocity (m/s)'],
                peakVelocity: rep['Peak Velocity (m/s)'],
                peakVelocityPerBW: rep['Peak Velocity / Body Weight (m/s/kg)'],
                
                meanForce: rep['Mean Force (N)'],
                meanForcePerBW: rep['Mean Force / Body Weight (N/kg)'],
                peakForce: rep['Peak Force (N)'],
                peakForcePerBW: rep['Peak Force / Body Weight (N/kg)'],
                
                meanPowerMPV: rep['Mean Power [MPV] (W)'],
                meanPowerMV: rep['Mean Power [MV] (W)'],
                meanPowerMPVPerBW: rep['Mean Power / Body Weight [MPV] (w/kg)'],
                meanPowerMVPerBW: rep['Mean Power / Body Weight [MV] (w/kg)'],
                peakPower: rep['Peak Power (W)'],
                peakPowerPerBW: rep['Peak Power / Body Weight (w/kg)'],
                
                meanAcceleration: rep['Mean Acceleration (m/s2)'],
                peakAcceleration: rep['Peak Acceleration (m/s2)'],
                accelerationIndex: rep['Acceleration Index (m/s2)'],
                
                fatigueIndexMPV: rep['Fatigue [MPV] (%)'],
                fatigueIndexMV: rep['Fatigue [MV] (%)'],
                fatigueIndexPV: rep['Fatigue (PV) (%)'],
                
                estimated1RM: rep['1RM (kg)'],
                estimated1RMPerBW: rep['1RM / Body Weight (%)'],
                
                rom: rep['ROM (Range of Motion) (m)'],
                repetitionDuration: rep['Repetition Duration (ms)'],
                timeToPeakVelocity: rep['Time to Peak Velocity (ms)'],
                
                timestamp: new Date(), // TODO: parse from Repetition Date
              });
              
              repsImported++;
            }
          }
        }
      });
      
      return {
        success: true,
        setsImported,
        repsImported,
        errors,
      };
    } catch (error) {
      return {
        success: false,
        setsImported,
        repsImported,
        errors: [...errors, error.message],
      };
    }
  }
}
```

```typescript
app.post('/api/vbt/upload', isAuthenticated, upload.single('file'), async (req: AuthRequest, res) => {
  try {
    if (!req.file) {

    // Validate file size (max 10MB)
    if (req.file.size > 10 * 1024 * 1024) {

    }
    
    // Validate file type
    if (!req.file.originalname.endsWith('.csv')) {
      return res.status(400).json({ message: 'Only CSV files accepted' });
    }
    
    // Import data
    const result = await VBTImporter.importVitruve(
      req.currentUser!.id,
      workoutSessionId,
      req.file.buffer,
      req.file.originalname
    );
    
    if (!result.success) {
      return res.status(400).json({
        message: 'Import failed',
        errors: result.errors,
      });
    }
    
    res.json({
      message: 'Import successful',
      setsImported: result.setsImported,
      repsImported: result.repsImported,
      errors: result.errors,
    });
  } catch (error) {
    console.error('VBT upload error:', error);
    res.status(500).json({ message: 'Failed to process upload' });
  }
});
```

```tsx
// client/src/components/VBTUpload.tsx

import { useState } from 'react';
import { useMutation } from '@tanstack/react-query';
import { Upload, CheckCircle, XCircle, Loader2 } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { useToast } from '@/hooks/use-toast';

interface VBTUploadProps {
  workoutSessionId: string;
  onSuccess?: () => void;
}

export function VBTUpload({ workoutSessionId, onSuccess }: VBTUploadProps) {
  const { toast } = useToast();
  const [file, setFile] = useState<File | null>(null);
  const [isDragging, setIsDragging] = useState(false);
  
  const uploadMutation = useMutation({
    mutationFn: async (file: File) => {
      const formData = new FormData();
      formData.append('file', file);
      formData.append('workoutSessionId', workoutSessionId);
      
      const res = await fetch('/api/vbt/upload', {
        method: 'POST',
        body: formData,
      });
      
      if (!res.ok) {
        const error = await res.json();
        throw new Error(error.message || 'Upload failed');
      }
      
      return res.json();
    },
    onSuccess: (data) => {
      toast({
        title: 'VBT Data Imported!',
        description: `${data.setsImported} sets and ${data.repsImported} reps imported successfully.`,
      });
      setFile(null);
      onSuccess?.();
    },
    onError: (error: Error) => {
      toast({
        title: 'Import Failed',
        description: error.message,
        variant: 'destructive',
      });
    },
  });
  
  const handleFileSelect = (selectedFile: File) => {
    if (!selectedFile.name.endsWith('.csv')) {
      toast({
        title: 'Invalid File',
        description: 'Please upload a CSV file',
        variant: 'destructive',
      });
      return;
    }
    
    if (selectedFile.size > 10 * 1024 * 1024) {
      toast({
        title: 'File Too Large',
        description: 'Maximum file size is 10MB',
        variant: 'destructive',
      });
      return;
    }
    
    setFile(selectedFile);
  };
  
  const handleDrop = (e: React.DragEvent) => {
    e.preventDefault();
    setIsDragging(false);
    
    const droppedFile = e.dataTransfer.files[0];
    if (droppedFile) {
      handleFileSelect(droppedFile);
    }
  };
  
  const handleUpload = () => {
    if (file) {
      uploadMutation.mutate(file);
    }
  };
  
  return (
    <Card>
      <CardHeader>
        <CardTitle className="flex items-center gap-2">
          <Upload className="h-5 w-5" />
          Import Velocity Data
        </CardTitle>
      </CardHeader>
      <CardContent>
        {/* Drag & Drop Zone */}
        <div
          className={`border-2 border-dashed rounded-lg p-8 text-center transition-colors ${
            isDragging ? 'border-primary bg-primary/10' : 'border-muted'
          }`}
          onDragOver={(e) => {
            e.preventDefault();
            setIsDragging(true);
          }}
          onDragLeave={() => setIsDragging(false)}
          onDrop={handleDrop}
        >
          {!file ? (
            <>
              <Upload className="h-12 w-12 mx-auto mb-4 text-muted-foreground" />
              <p className="text-sm text-muted-foreground mb-2">
                Drag & drop your Vitruve CSV file here, or
              </p>
              <label>
                <Button variant="outline" asChild>
                  <span>
                    <input
                      type="file"
                      accept=".csv"
                      className="hidden"
                      onChange={(e) => {
                        const selectedFile = e.target.files?.[0];
                        if (selectedFile) handleFileSelect(selectedFile);
                      }}
                    />
                    Browse Files
                  </span>
                </Button>
              </label>
              <p className="text-xs text-muted-foreground mt-4">
                Supports: Vitruve, GymAware, and custom CSV formats (max 10MB)
              </p>
            </>
          ) : (
            <>
              <CheckCircle className="h-12 w-12 mx-auto mb-4 text-green-500" />
              <p className="font-medium mb-1">{file.name}</p>
              <p className="text-sm text-muted-foreground mb-4">
                {(file.size / 1024).toFixed(1)} KB
              </p>
              <div className="flex gap-2 justify-center">
                <Button
                  onClick={handleUpload}
                  disabled={uploadMutation.isPending}
                >
                  {uploadMutation.isPending && (
                    <Loader2 className="mr-2 h-4 w-4 animate-spin" />
                  )}
                  Upload & Import
                </Button>
                <Button
                  variant="outline"
                  onClick={() => setFile(null)}
                  disabled={uploadMutation.isPending}
                >
                  Cancel
                </Button>
              </div>
            </>
          )}
        </div>
        
        {/* Import Status */}
        {uploadMutation.isSuccess && (
          <div className="mt-4 p-4 bg-green-50 dark:bg-green-900/20 border border-green-200 dark:border-green-800 rounded-lg">
            <p className="text-sm text-green-800 dark:text-green-200">
              ✓ Import completed successfully
            </p>
          </div>
        )}
        
        {uploadMutation.isError && (
          <div className="mt-4 p-4 bg-red-50 dark:bg-red-900/20 border border-red-200 dark:border-red-800 rounded-lg">
            <p className="text-sm text-red-800 dark:text-red-200">
              {uploadMutation.error?.message}
            </p>
          </div>
        )}
      </CardContent>
    </Card>
  );
}
```
